// 열에 쓰는 대신 즉시 에러를 던진다. (연 단위 자동 생성 시 특히 중요 — 템플릿에 발주 열이

function assertLedgerLayout_(ledgerSheet) {
  const lastColumn = Math.max(ledgerSheet.getLastColumn(), LEDGER_COL.FILE_LINK);
  const headers = ledgerSheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0];
  const orderHeader = String(headers[LEDGER_COL.ORDER - 1] || '');
  const quoteNumberHeader = String(headers[LEDGER_COL.QUOTE_NUMBER - 1] || '');
  if (orderHeader.indexOf('발주') === -1 || quoteNumberHeader.indexOf('견적번호') === -1) {
    throw new Error(
      '견적관리대장의 열 구성이 예상과 다릅니다(E열=발주, F열=견적번호이어야 합니다). ' +
      'LEDGER_TEMPLATE_ID 대장 템플릿 또는 이 연도 시트의 헤더 행을 확인해 주세요.'
    );
  }
  return ledgerSheet;
}

function getQuoteLedgerFromReact(requestedYear, requestedDepartment) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '접속 권한이 없습니다.' };
    }

    // 관리자는 모든 부서의 대장을 선택해 조회할 수 있다.
    // 일반 사용자는 본인 등록 부서의 대장만 조회할 수 있다.
    let department = access.author.department;
    const targetDept = String(requestedDepartment || '').trim();
    if (access.isAdmin && targetDept && CONFIG.DEPARTMENT_ROOTS[targetDept]) {
      department = targetDept;
    }

    const rootFolderId = resolveRootFolderId({ authorDepartment: department });
    const currentYear = new Date().getFullYear();
    const parsedYear = Number(requestedYear);
    const year = Number.isInteger(parsedYear) && parsedYear >= 2000 && parsedYear <= currentYear
      ? parsedYear
      : currentYear;
    const availableYears = getAvailableLedgerYears_(rootFolderId, currentYear);
    const system = getExistingYearSystem_(year, rootFolderId);
    const baseResult = {
      success: true,
      availableYears: availableYears,
      department: department,
      isAdmin: Boolean(access.isAdmin),
      availableDepartments: Object.keys(CONFIG.DEPARTMENT_ROOTS)
    };
    if (!system) {
      return Object.assign({ headers: [], rows: [] }, baseResult);
    }
    const sheet = system.ledgerSheet;
    const lastRow = sheet.getLastRow();
    const lastColumn = sheet.getLastColumn();

    if (lastRow < 1 || lastColumn < 1) {
      return Object.assign({ headers: [], rows: [] }, baseResult);
    }

    const range = sheet.getRange(1, 1, lastRow, lastColumn);
    const displayValues = range.getDisplayValues();
    const richTextValues = range.getRichTextValues();
    const headers = displayValues[0];
    const rows = displayValues.slice(1).map(function(values, rowIndex) {
      return {
        values: values,
        links: values.map(function(_, columnIndex) {
          const richText = richTextValues[rowIndex + 1][columnIndex];
          return richText ? richText.getLinkUrl() : null;
        })
      };
    });

    // 파일 생성이 완료 보고 반영보다 늦더라도, 목록 조회 시점에 누락된 파일링크를 즉시 복구한다.
    // (비동기 완료 보고를 기다리지 않고, 매 조회마다 빈 링크 행만 스냅샷 기반으로 다시 채운다.)
    repairEmptyLedgerLinks_(system, department, year, rows);

    return Object.assign({ headers: headers, rows: rows }, baseResult);
  } catch (err) {
    Logger.log('대장 조회 오류: ' + err.toString());
    return {
      success: false,
      message: '해당 부서 대장을 조회하지 못했습니다: ' + err.toString()
    };
  }
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    // Vercel 폴백 경로(주로 사용되지 않음) — Apps Script 프로젝트의 스크립트 속성(Script Properties)에
    // AGENT_TOKEN 값을 설정해야만 이 경로가 동작한다. 설정되어 있지 않으면 무조건 거부한다(fail-closed).
    // 설정 방법: Apps Script 편집기 > 프로젝트 설정 > 스크립트 속성 > AGENT_TOKEN 추가,
    // 그리고 Vercel 환경변수 APPS_SCRIPT_AGENT_TOKEN에 같은 값을 설정한다.
    const expectedToken = String(PropertiesService.getScriptProperties().getProperty('AGENT_TOKEN') || '');
    if (!expectedToken || String(payload.agentToken || '') !== expectedToken) {
      return jsonResponse_({ success: false, message: '토큰 인증에 실패했습니다.' });
    }
    const result = processQuoteFromReact(payload);
    return jsonResponse_(result);
  } catch (err) {
    return jsonResponse_({ success: false, message: err.toString() });
  }
}

function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ── 사내 PC 로컬 저장 에이전트 (Google Drive 폴더 연동) ───────────────────
// Apps Script는 대기 폴더에 견적 JSON 파일만 쓰고, 사내 PC의 에이전트가
// Google Drive 데스크톱 동기화로 파일을 받아 XLSX/PDF를 생성·저장한다.
// 에이전트는 완료 보고를 results 폴더에 쓰며, 1분 트리거가 대장 파일링크를 갱신한다.

// 최신 견적관리대장 사본(XLSX)을 에이전트 전달 폴더에 내보낸다.
// 에이전트가 견적 파일 생성 시 같은 연도 폴더에 함께 복사한다.
function exportLedgerToAgent_(system, year, quoteNumber) {
  try {
    const ledgerId = system.ledger.getId();
    const res = UrlFetchApp.fetch('https://docs.google.com/spreadsheets/d/' + ledgerId + '/export?format=xlsx', {
      method: 'get',
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      muteHttpExceptions: true
    });
    if (res.getResponseCode() !== 200) {
      Logger.log('대장 내보내기 실패: ' + res.getContentText());
      return;
    }
    const delivery = getAgentSubfolder_('delivery');
    const fileName = `대장_${sanitizeDriveName(quoteNumber)}.xlsx`;
    const blob = res.getBlob().setName(fileName);
    delivery.createFile(blob);
  } catch (err) {
    Logger.log('대장 사본 전달 오류: ' + err.toString());
  }
}

function getAgentFolder_() {
  const props = PropertiesService.getScriptProperties();
  const cachedId = props.getProperty('AGENT_FOLDER_ID');
  if (cachedId) {
    try {
      const folder = DriveApp.getFolderById(cachedId);
      if (folder && !folder.isTrashed()) return folder;
    } catch (err) { /* 폴더가 삭제되었거나 접근 불가 시 재탐색 */ }
  }
  const authorFile = DriveApp.getFileById(CONFIG.AUTHOR_DB_SHEET_ID);
  const parents = authorFile.getParents();
  const parent = parents.hasNext() ? parents.next() : DriveApp.getRootFolder();
  let folder = null;
  const iter = parent.getFoldersByName(CONFIG.AGENT_QUEUE_FOLDER_NAME);
  while (iter.hasNext()) {
    const candidate = iter.next();
    if (!candidate.isTrashed()) {
      folder = candidate;
      break;
    }
  }
  if (!folder) {
    folder = parent.createFolder(CONFIG.AGENT_QUEUE_FOLDER_NAME);
  }
  props.setProperty('AGENT_FOLDER_ID', folder.getId());
  return folder;
}

function getAgentSubfolder_(name) {
  const root = getAgentFolder_();
  const iter = root.getFoldersByName(name);
  while (iter.hasNext()) {
    const candidate = iter.next();
    if (!candidate.isTrashed()) return candidate;
  }
  return root.createFolder(name);
}

// 로컬 LOG 폴더(계정별/일자별)에 기록할 활동 로그를 Drive 동기화 폴더를 통해 에이전트로 전달한다.
// 브라우저는 HTTPS 페이지에서 사내 http 서버로 fetch할 수 없으므로 파일 큐를 경유한다.

  try {
    const logs = getAgentSubfolder_('logs');

    logs.createFile(fileName, JSON.stringify({ account: account || 'unknown', eventType: eventType, detail: detail || '' }), MimeType.PLAIN_TEXT);
  } catch (err) {
    Logger.log('활동 로그 큐 작성 실패: ' + err.toString());
  }
}

function enqueueLocalSave_(details, items, vatTotal, createDraft, customSubject) {
  const pending = getAgentSubfolder_('pending');
  const fileName = `${sanitizeDriveName(details.quoteNumber)}.json`;
  const payload = JSON.stringify({
    details: details,
    items: items || [],
    vatTotal: vatTotal,
    createDraft: Boolean(createDraft),
    customSubject: customSubject || ''
  });
  pending.createFile(fileName, payload, MimeType.PLAIN_TEXT);
  ensureAgentTrigger_();
}

function splitQuoteItems_(items) {
  const parts = [];
  for (let i = 0; i < items.length; i += ITEMS_PER_FILE) {
    parts.push(items.slice(i, i + ITEMS_PER_FILE));
  }
  return parts.length > 0 ? parts : [[]];
}

function quotePartNumber_(baseQuoteNumber, partIndex, partCount) {
  return partCount > 1 ? `${baseQuoteNumber}_${partIndex + 1}` : baseQuoteNumber;
}

function ensureAgentTrigger_() {
  const exists = ScriptApp.getProjectTriggers().some(function(t) {
    return t.getHandlerFunction() === 'processAgentResults';
  });
  if (!exists) {
    ScriptApp.newTrigger('processAgentResults').timeBased().everyMinutes(1).create();
  }
}

// 에이전트가 XLSX/PDF 생성에 성공한 뒤에만 대장 행을 기록한다.
// 결과 JSON에 원본 상세 데이터가 포함되어 있어 재시도해도 동일한 행/스냅샷을 중복 생성하지 않는다.
function commitLedgerAfterAgentSuccess_(result) {
  if (!result || !result.details || !Array.isArray(result.items)) {
    throw new Error('완료 보고에 대장 저장용 견적 상세 데이터가 없습니다. 에이전트 최신 코드를 반영해 주세요.');
  }
  const rootFolderId = resolveRootFolderId({ authorDepartment: result.department });
  const year = Number(result.year) || new Date().getFullYear();
  const system = getExistingYearSystem_(year, rootFolderId);
  if (!system) throw new Error('대장을 찾을 수 없습니다(부서: ' + result.department + ', 연도: ' + year + ')');

  result.details.authorDepartment = result.department;
  result.details.year = year;
  result.details.quoteNumber = result.quoteNumber;
  if (result.ledgerQuoteNumber && result.revisionNumber > 0) {
    result.details.baseQuoteNumber = result.ledgerQuoteNumber;
    result.details.revisionNumber = result.revisionNumber;
  }
  saveToLedgerSheet(
    system.ledgerSheet,
    result.details,
    result.items,
    Number(result.vatTotal || result.details.totalPrice || 0),
    ''
  );
  return system;
}

// 1분마다 실행: 에이전트가 쓴 완료 보고를 반영해 대장 파일링크를 갱신하고 잔여 파일을 정리한다.
function processAgentResults() {
  let results;
  try {
    results = getAgentSubfolder_('results');
  } catch (err) {
    return;
  }
  const files = results.getFiles();
  while (files.hasNext()) {
    const file = files.next();
    let result = null;
    try {
      result = JSON.parse(file.getBlob().getDataAsString('utf8'));
    } catch (err) {
      continue; // 아직 동기화 중인 파일 — 다음 트리거에서 재시도
    }

    if (result && result.ok && result.department && result.quoteNumber && result.fileUrl) {
      try {
        // 파일 생성 성공을 먼저 확인한 뒤에만 대장 행을 만들고, 그 다음 링크를 기록한다.
        const system = commitLedgerAfterAgentSuccess_(result);
        exportLedgerToAgent_(system, Number(result.year), result.quoteNumber);
        updateLedgerLink_(result);
        Logger.log('대장 파일링크 갱신 완료(' + result.quoteNumber + '): ' + result.fileUrl);
      } catch (err) {
        // 파일은 이미 생성되었지만 대장 링크 갱신에 실패한 상태 — 결과를 지우면 영구히 유실되므로
        // resultsFailed 폴더에 원본을 보존하고, 매분 트리거가 같은 오류로 계속 재시도하지 않도록
        // results 폴더에서는 제거한다. 관리자가 수동으로 확인/재처리해야 한다.
        Logger.log('대장 파일링크 갱신 실패(' + result.quoteNumber + '): ' + err.toString());
        try {
          getAgentSubfolder_('resultsFailed').createFile(
            `링크실패_${sanitizeDriveName(result.quoteNumber || file.getName())}_${Date.now()}.json`,
            JSON.stringify({ result: result, error: err.toString() }),
            MimeType.PLAIN_TEXT
          );
        } catch (copyErr) {
          Logger.log('resultsFailed 보존 실패: ' + copyErr.toString());
        }
      }
    } else if (result && result.ok) {
      // 성공 보고인데 필수 값이 빠진 경우에는 조용히 삭제하지 않는다.
      // fileUrl이 비어 있으면 대장 링크를 만들 수 없으므로 원본 결과를 보존해야 한다.
      const missing = [];
      if (!result.department) missing.push('department');
      if (!result.quoteNumber) missing.push('quoteNumber');
      if (!result.fileUrl) missing.push('fileUrl');
      const message = '완료 보고 필수 값 누락: ' + missing.join(', ');
      Logger.log('대장 파일링크 갱신 보류(' + (result.quoteNumber || file.getName()) + '): ' + message);
      try {
        getAgentSubfolder_('resultsFailed').createFile(
          `링크실패_${sanitizeDriveName(result.quoteNumber || file.getName())}_${Date.now()}.json`,
          JSON.stringify({ result: result, error: message }),
          MimeType.PLAIN_TEXT
        );
      } catch (copyErr) {
        Logger.log('resultsFailed 보존 실패: ' + copyErr.toString());
      }
    } else if (result && result.ok === false) {
      movePendingToFailed_(result.quoteNumber, result.error || '알 수 없는 오류');
    }
    permanentlyDeleteFile_(file.getId());
  }
}

function updateLedgerLink_(result) {
  const rootFolderId = resolveRootFolderId({ authorDepartment: result.department });
  const year = Number(result.year) || new Date().getFullYear();
  const system = getExistingYearSystem_(year, rootFolderId);
  if (!system) {
    throw new Error('대장을 찾을 수 없습니다(부서: ' + result.department + ', 연도: ' + year + ')');
  }
  const ledger = system.ledgerSheet;
  const ledgerQuoteNumber = String(result.ledgerQuoteNumber || baseQuoteNumber_(result.quoteNumber)).trim();
  const row = findLedgerRow_(ledger, ledgerQuoteNumber);
  if (row < 2) {
    throw new Error('대장에서 견적번호를 찾을 수 없습니다: ' + ledgerQuoteNumber);
  }

  const fileUrl = String(result.fileUrl);
  const path = fileUrl.split('?')[0].split('/');
  const fileName = decodeURIComponent(path[path.length - 1] || '열기');
  const revisionNumber = Number(result.revisionNumber) || 0;
  const linkLabel = revisionNumber > 0
    ? `Rev${String(revisionNumber).padStart(2, '0')}: ${fileName}`
    : `원본: ${fileName}`;
  const cell = ledger.getRange(row, LEDGER_COL.FILE_LINK);
  const links = [];
  const existing = cell.getRichTextValue();
  if (existing) {
    existing.getRuns().forEach(function(run) {
      const url = run.getLinkUrl();
      if (!url) return; // 구분자(' | ')처럼 링크가 없는 런은 누적하지 않는다.
      const text = run.getText();
      if (text) links.push({ text: text, url: url });
    });
  }
  if (links.length === 0) {
    const currentText = String(cell.getDisplayValue() || '').trim();
    if (currentText) links.push({ text: currentText, url: null });
  }
  if (links.some(function(link) { return link.url === fileUrl; })) return;
  links.push({ text: linkLabel, url: fileUrl });

  const separator = ' | ';
  const combinedText = links.map(function(link) { return link.text; }).join(separator);
  const builder = SpreadsheetApp.newRichTextValue().setText(combinedText);
  let offset = 0;
  links.forEach(function(link) {
    if (link.url) builder.setLinkUrl(offset, offset + link.text.length, link.url);
    offset += link.text.length + separator.length;
  });
  cell.setRichTextValue(builder.build());
}

function movePendingToFailed_(quoteNumber, error) {
  try {
    const pending = getAgentSubfolder_('pending');
    const failed = getAgentSubfolder_('failed');
    const iter = pending.getFilesByName(`${sanitizeDriveName(quoteNumber)}.json`);
    if (iter.hasNext()) {
      const file = iter.next();
      failed.createFile(
        `실패_${sanitizeDriveName(quoteNumber)}_${Date.now()}.json`,
        JSON.stringify({ quoteNumber: quoteNumber, error: error }),
        MimeType.PLAIN_TEXT
      );
      file.setTrashed(true);
    }
  } catch (err) {
    Logger.log('실패 폴더 이동 오류(' + quoteNumber + '): ' + err.toString());
  }
}

// 작성자 DB 시트(시트1)에서 행 목록을 읽는 공용 헬퍼.
// 기대하는 열 구성: 작성자(또는 이름/성명) | 연락처 | 이메일 | 부서
// 1행 헤더의 키워드로 열 위치를 찾고, 헤더가 없으면 A~D열 고정 순서로 해석한다.
function readAuthorRows_() {
  const spreadsheet = SpreadsheetApp.openById(CONFIG.AUTHOR_DB_SHEET_ID);
  const sheet = spreadsheet.getSheetByName(CONFIG.AUTHOR_DB_SHEET_NAME) || spreadsheet.getSheets()[0];
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  if (lastRow < 1 || lastColumn < 1) return [];

  const values = sheet.getRange(1, 1, lastRow, lastColumn).getValues();
  const headers = values[0].map(function(v) { return String(v || '').trim(); });

  function findColumn(keywords) {
    for (let c = 0; c < headers.length; c++) {
      for (const keyword of keywords) {
        if (headers[c].indexOf(keyword) !== -1) return c;
      }
    }
    return -1;
  }

  let nameCol = findColumn(['작성자', '이름', '성명']);
  let phoneCol = findColumn(['연락처', '전화', '휴대폰']);
  let emailCol = findColumn(['이메일', '메일']);
  let deptCol = findColumn(['부서', '소속', '팀']);

  const headerMatched = nameCol !== -1;
  if (!headerMatched) {
    nameCol = 0;
    phoneCol = 1;
    emailCol = 2;
    deptCol = 3;
  }

  const rows = [];
  for (let r = headerMatched ? 1 : 0; r < values.length; r++) {
    const name = String(values[r][nameCol] || '').trim();
    if (!name) continue;
    rows.push({
      name: name,
      phone: phoneCol >= 0 ? String(values[r][phoneCol] || '').trim() : '',
      email: emailCol >= 0 ? String(values[r][emailCol] || '').trim() : '',
      department: deptCol >= 0 ? String(values[r][deptCol] || '').trim() : ''
    });
  }
  return rows;
}

      availableDepartments: Object.keys(CONFIG.DEPARTMENT_ROOTS)
    };

  }
}

function getYearSystem(year, rootFolderId) {
  const targetRootId = rootFolderId || CONFIG.DEPARTMENT_ROOTS[CONFIG.DEFAULT_DEPARTMENT];
  const rootFolder = DriveApp.getFolderById(targetRootId);
  const yearStr = year.toString();

  let yearFolder;
  const folderIter = rootFolder.getFoldersByName(yearStr);
  if (folderIter.hasNext()) {
    yearFolder = folderIter.next();
  } else {
    yearFolder = rootFolder.createFolder(yearStr);
  }

  const ledgerName = `${yearStr}_견적관리대장`;
  let ledgerSS;
  const fileIter = yearFolder.getFilesByName(ledgerName);
  if (fileIter.hasNext()) {
    ledgerSS = SpreadsheetApp.open(fileIter.next());
  } else {
    // 새 연도가 시작되면 "견적관리대장 샘플" 템플릿을 복사해 해당 연도 폴더에 만든다.
    const ledgerTemplate = DriveApp.getFileById(CONFIG.LEDGER_TEMPLATE_ID);
    const newLedgerFile = ledgerTemplate.makeCopy(ledgerName, yearFolder);
    ledgerSS = SpreadsheetApp.open(newLedgerFile);
  }

  const ledgerSheet = ledgerSS.getSheetByName(CONFIG.LEDGER_SHEET_NAME) || ledgerSS.getSheets()[0];
  return { folder: yearFolder, ledger: ledgerSS, ledgerSheet };
}

// 목록 조회에서는 존재하지 않는 연도의 폴더·대장을 새로 만들지 않는다.
function getExistingYearSystem_(year, rootFolderId) {
  const targetRootId = rootFolderId || CONFIG.DEPARTMENT_ROOTS[CONFIG.DEFAULT_DEPARTMENT];
  const rootFolder = DriveApp.getFolderById(targetRootId);
  const yearStr = String(year);
  const folderIter = rootFolder.getFoldersByName(yearStr);
  if (!folderIter.hasNext()) return null;
  const yearFolder = folderIter.next();
  const ledgerName = `${yearStr}_견적관리대장`;
  const fileIter = yearFolder.getFilesByName(ledgerName);
  if (!fileIter.hasNext()) return null;
  const ledgerSS = SpreadsheetApp.open(fileIter.next());
  const ledgerSheet = ledgerSS.getSheetByName(CONFIG.LEDGER_SHEET_NAME) || ledgerSS.getSheets()[0];
  return { folder: yearFolder, ledger: ledgerSS, ledgerSheet };
}

function getAvailableLedgerYears_(rootFolderId, currentYear) {
  const targetRootId = rootFolderId || CONFIG.DEPARTMENT_ROOTS[CONFIG.DEFAULT_DEPARTMENT];
  const rootFolder = DriveApp.getFolderById(targetRootId);
  const folders = rootFolder.getFolders();
  const years = [];
  while (folders.hasNext()) {
    const folder = folders.next();
    const yearName = String(folder.getName()).trim();
    if (!/^\d{4}$/.test(yearName)) continue;
    const year = Number(yearName);
    if (year > currentYear) continue;
    const ledgerFiles = folder.getFilesByName(`${yearName}_견적관리대장`);
    if (ledgerFiles.hasNext()) years.push(year);
  }
  return years.sort((left, right) => right - left);
}

function baseQuoteNumber_(quoteNumber) {
  return String(quoteNumber || '').trim().replace(/_Rev\d+$/i, '');
}

function findLedgerRow_(sheet, quoteNumber) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;
  const numbers = sheet.getRange(2, LEDGER_COL.QUOTE_NUMBER, lastRow - 1, 1).getDisplayValues();
  const target = String(quoteNumber || '').trim();
  for (let i = 0; i < numbers.length; i++) {
    if (String(numbers[i][0] || '').trim() === target) return i + 2;
  }
  return -1;
}

function getNextRevisionNumber_(spreadsheet, ledgerSheet, baseQuoteNumber) {
  let maxRevision = 0;
  const snapshot = getQuoteSnapshotSheet_(spreadsheet, false);
  if (snapshot && snapshot.getLastRow() >= 2) {
    const rows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 2).getDisplayValues();
    rows.forEach(function(row) {
      if (String(row[1] || '').trim() !== baseQuoteNumber) return;
      const match = String(row[0] || '').match(/_Rev(\d+)$/i);
      if (match) maxRevision = Math.max(maxRevision, parseInt(match[1], 10));
    });
  }

  const ledgerRow = findLedgerRow_(ledgerSheet, baseQuoteNumber);
  if (ledgerRow >= 2) {
    const linkText = String(ledgerSheet.getRange(ledgerRow, LEDGER_COL.FILE_LINK).getDisplayValue() || '');
    const matches = linkText.match(/_Rev(\d+)/gi) || [];
    matches.forEach(function(value) {
      const match = value.match(/(\d+)/);
      if (match) maxRevision = Math.max(maxRevision, parseInt(match[1], 10));
    });
  }
  return maxRevision + 1;
}

  const outputQuoteNumber = String(details.quoteNumber || '').trim();

  // 완료 보고 재처리 시 같은 스냅샷을 중복으로 쌓지 않는다.
  if (snapshot.getLastRow() >= 2) {

    outputQuoteNumber,

    new Date(),
    String(details.authorEmail || '').trim(),

  ]);

function updateQuoteOrderFromReact(payload) {

      return { success: false, message: '발주 상태를 변경할 권한이 없습니다.' };
    }
    const currentYear = new Date().getFullYear();
    const parsedYear = Number(payload && payload.year);
    const year = Number.isInteger(parsedYear) && parsedYear >= 2000 && parsedYear <= currentYear
      ? parsedYear
      : currentYear;

    const baseQuoteNumber = baseQuoteNumber_(payload && payload.quoteNumber);
    const prefixDept = String(baseQuoteNumber || '').trim().split(/\s+/)[0];
    let department = access.author.department;
    if (CONFIG.DEPARTMENT_ROOTS[prefixDept]) {
      department = prefixDept;
    } else if (access.isAdmin && payload && payload.department && CONFIG.DEPARTMENT_ROOTS[String(payload.department).trim()]) {
      department = String(payload.department).trim();
    }

    const rootFolderId = resolveRootFolderId({ authorDepartment: department });
    const system = getExistingYearSystem_(year, rootFolderId);

    // 대장 구조가 예상과 다를 때 엉뚱한 열을 덮어쓰지 않도록 검증 실패 시 전체 헤더에서 재탐색한다.
    const headers = system.ledgerSheet.getRange(1, 1, 1, system.ledgerSheet.getLastColumn()).getDisplayValues()[0];
    let orderColumn = String(headers[LEDGER_COL.ORDER - 1] || '').indexOf('발주') !== -1 ? LEDGER_COL.ORDER - 1 : -1;
    if (orderColumn < 0) {
      orderColumn = headers.findIndex(function(header) {
        return String(header || '').indexOf('발주') !== -1;
      });

    if (orderColumn < 0) {
      return { success: false, message: '견적관리대장에 발주 컬럼이 없습니다.' };
    }
    system.ledgerSheet.getRange(row, orderColumn + 1).setValue(payload && payload.ordered === true ? '발주' : '');
    enqueueActivityLog_(
      access.author.email || access.author.name || department,
      '발주',
      `견적번호: ${baseQuoteNumber} | 상태: ${(payload && payload.ordered === true) ? '발주 완료(체크)' : '발주 취소(체크 해제)'} | 부서: ${department}`
    );
    return { success: true, message: '발주 여부가 기록되었습니다.' };
  } catch (err) {
    return { success: false, message: '발주 상태 기록 실패: ' + err.toString() };
  }
}

// 발주등록 요청 메일 초안 생성 (현재 로그인한 담당자의 Gmail 임시보관함에 저장)
// payload: {
//   year, department, quoteNumber,
//   clientName, productName, contactName, contactPhone,
//   deliveryAddress,   // 납품 주소 (필수)
//   files: [ { name, mimeType, base64 } ]  // 1개 이상 필수 (발주서·사업자등록증 등)
// }
// 발주등록 요청 메일 초안 생성 (현재 로그인한 담당자의 Gmail 임시보관함에 저장)
// payload.files는 로컬 에이전트 order-email 페이지가 읽어 postMessage로 전달한 base64 첨부 목록이다.
function createOrderDraftFromReact(payload) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '발주등록 요청 메일 작성 권한이 없습니다.' };
    }

    const clientName = String((payload && payload.clientName) || '').trim();
    const productName = String((payload && payload.productName) || '').trim();
    const contactName = String((payload && payload.contactName) || '').trim();
    const contactPhone = String((payload && payload.contactPhone) || '').trim();
    const deliveryAddress = String((payload && payload.deliveryAddress) || '').trim();
    const files = Array.isArray(payload && payload.files)
      ? payload.files
      : [];

    if (!deliveryAddress) {
      return { success: false, message: '납품 주소를 입력해야 발주 메일을 작성할 수 있습니다.' };
    }
    if (files.length === 0) {
      return { success: false, message: '첨부할 파일(발주서, 사업자등록증 등)을 1개 이상 선택해 주세요.' };
    }
    if (files.length > 10) {
      return { success: false, message: '첨부 파일은 최대 10개까지 선택할 수 있습니다.' };
    }

    const quoteNumber = String((payload && payload.quoteNumber) || '').trim();
    const attachments = [];
    files.forEach(function(file) {
      const fileName = String(file && file.name || '첨부파일');
      const mimeType = String(file && file.mimeType || 'application/octet-stream');
      const base64 = String(file && file.base64 || '').replace(/[^A-Za-z0-9+/=]/g, '');
      if (!base64) throw new Error('첨부 파일 데이터가 비어 있습니다: ' + fileName);
      try {
        // Apps Script는 parseUint8Array를 제공하지 않으므로 base64Decode로 Blob 바이트를 만든다.
        const bytes = Utilities.base64Decode(base64);
        attachments.push(Utilities.newBlob(bytes, mimeType).setName(fileName));
      } catch (err) {
        throw new Error('첨부 파일 변환 실패(' + fileName + '): ' + err.toString());
      }
    });

    const subject = `[발주서] ${clientName}- ${productName} 발주서 공유`.trim();

    // &부서 표기: 기술영업→기술영업실, 영업→자동화영업실, 프로젝트→프로젝트사업실
    const deptName = String((payload && payload.department) || access.author.department || '').trim();
    const deptLabelMap = { '기술영업': '기술영업실', '영업': '자동화영업실', '프로젝트': '프로젝트사업실' };
    const deptLabel = deptLabelMap[deptName] || (deptName ? deptName + '실' : '기술영업실');

    const body = [
      '안녕하세요,',
      `${deptLabel} ${access.author.name || ''}입니다.`,
      '',
      `${clientName} ${productName} 발주서를 공유드립니다.`,
      '',
      '<납품정보>',
      '',
      `주소: ${deliveryAddress}`,
      `수령인: ${contactName}`,
      `연락처: ${contactPhone}`,
      '',
      '필요하신 추가 사항이 있으시면 연락 또는 회신 부탁드립니다.',
      '',
      '감사합니다.'
    ].join('\n');

    // 본인(로그인 사용자) 계정으로 초안을 생성한다.
    GmailApp.createDraft(ORDER_EMAIL_TO, subject, body, {
      cc: ORDER_EMAIL_CC,
      attachments: attachments,
      name: 'CIMON'
    });

    enqueueActivityLog_(
      access.author.email || access.author.name || '',
      '발주 메일',
      `견적번호: ${quoteNumber} | 업체명: ${clientName} | 주소: ${deliveryAddress} | 첨부 ${attachments.length}개 | 임시보관함 초안 생성`
    );

    return {
      success: true,
      message: '발주등록 요청 메일 초안이 임시보관함에 생성되었습니다. 최종 확인 후 발송해 주세요.'
    };
  } catch (err) {
    return { success: false, message: '발주등록 요청 메일 작성 실패: ' + err.toString() };
  }
}

// 발주등록 요청 메일 첨부용 파일 목록 조회 (구글 드라이브 '문서/<견적폴더명>' 미러 기준)
// payload: { year, department, quoteNumber, company }
function getQuoteFilesFromReact(payload) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '발주등록 요청 메일 사용 권한이 없습니다.' };
    }
    const quoteNumber = String((payload && payload.quoteNumber) || '').trim();
    const company = String((payload && payload.company) || '').trim();
    if (!quoteNumber) {
      return { success: true, files: [], message: '견적번호가 없습니다.' };
    }

    let docsRoot;
    try {
      docsRoot = getAgentSubfolder_('문서');
    } catch (driveErr) {
      return {
        success: false,
        message: '구글 드라이브 공유 폴더(견적에이전트) 접근 권한이 없습니다. 관리자에게 공유 폴더 접근 권한을 요청해 주세요. (' + driveErr.toString() + ')'
      };
    }
    const folderName = `${baseQuoteNumber_(quoteNumber)}_${company}`;
    let docsFolder = null;
    const folderIter = docsRoot.getFoldersByName(folderName);
    if (folderIter.hasNext()) {
      docsFolder = folderIter.next();
    } else {
      const prefix = baseQuoteNumber_(quoteNumber) + '_';
      const allFolders = docsRoot.getFolders();
      while (allFolders.hasNext()) {
        const candidate = allFolders.next();
        if (String(candidate.getName() || '').startsWith(prefix)) {
          docsFolder = candidate;
          break;
        }
      }
    }
    if (!docsFolder) {
      return { success: true, files: [], folder: folderName };
    }

    const files = [];
    const fileIter = docsFolder.getFiles();
    while (fileIter.hasNext()) {
      const file = fileIter.next();
      if (file.isTrashed()) continue;
      let size = 0;
      try { size = file.getBlob().getSize(); } catch (err) { size = 0; }
      files.push({ name: file.getName(), size: size });
    }
    files.sort(function(a, b) { return String(a.name).localeCompare(String(b.name), 'ko'); });
    return { success: true, folder: docsFolder.getName(), files: files };
  } catch (err) {
    return { success: false, message: '견적 폴더 파일 목록 조회 실패: ' + err.toString() };
  }
}

// 대시보드 통계 데이터 조회 — 에이전트가 Drive '견적에이전트/stats/<부서>.json'에 기록한
// 경량 통계 JSON을 읽어 프론트에 반환한다. (견적 원본 파일은 Drive에 올리지 않음)
// payload: { department } — 빈 문자열이면 본인 부서, 관리자는 특정 부서 선택 가능
function getDashboardStatsFromReact(payload) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '대시보드 사용 권한이 없습니다.' };
    }
    let department = access.author.department || CONFIG.DEFAULT_DEPARTMENT;
    const requestedDept = String((payload && payload.department) || '').trim();
    if (access.isAdmin && requestedDept && CONFIG.DEPARTMENT_ROOTS[requestedDept]) {
      department = requestedDept;
    }

    const statsRoot = getAgentSubfolder_('stats');
    const fileIter = statsRoot.getFilesByName(`${department}.json`);
    if (!fileIter.hasNext()) {
      return { success: true, department, generatedAt: '', records: [] };
    }
    const file = fileIter.next();
    const parsed = JSON.parse(file.getBlob().getDataAsString('utf8'));
    return {
      success: true,
      department,
      generatedAt: String(parsed.generatedAt || ''),
      records: Array.isArray(parsed.records) ? parsed.records : [],
    };
  } catch (err) {
    return { success: false, message: '대시보드 통계 조회 실패: ' + err.toString() };
  }
}

function getNextQuoteNumber(rootFolderId, department) {
  try {
    const today = new Date();
    const system = getYearSystem(today.getFullYear(), rootFolderId);
    const sheet = system.ledgerSheet;
    const departmentName = String(department || CONFIG.DEFAULT_DEPARTMENT).trim() || CONFIG.DEFAULT_DEPARTMENT;
    const yearPrefix = `${departmentName} ${Utilities.formatDate(today, Session.getScriptTimeZone(), 'yy')}`;
    const lastRow = sheet.getLastRow();

    // 대장 행은 파일 생성 완료 후 기록되므로, Script Properties로 번호를 먼저 예약한다.
    // 그렇지 않으면 동시에 저장한 두 작업이 같은 번호를 받을 수 있다.
    let seq = 1;
    if (lastRow > 1) {
      const nos = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
      nos.forEach(function(row) {
        const value = Number(row[0]);
        if (Number.isInteger(value) && value >= seq) seq = value + 1;
      });
    }

    const reservationKey = `QUOTE_SEQ_${sanitizeDriveName(departmentName)}_${today.getFullYear()}`;
    const properties = PropertiesService.getScriptProperties();
    const reservedNext = Number(properties.getProperty(reservationKey) || 1);
    if (Number.isInteger(reservedNext) && reservedNext > seq) seq = reservedNext;
    properties.setProperty(reservationKey, String(seq + 1));

    const seqStr = String(seq).padStart(3, '0');
    return `${yearPrefix}${Utilities.formatDate(today, Session.getScriptTimeZone(), 'MM')}-${seqStr}`;
  } catch (err) {
    throw new Error('견적번호 생성 실패: ' + err.toString());
  }
}

function processQuote(quoteData, createDraft, customSubject, customBody, revisionOf, revisionYear, revisionDepartment) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
  } catch (err) {
    return { success: false, message: '다른 사용자가 저장 중입니다. 잠시 후 다시 시도해 주세요.' };
  }

  try {
    const details = quoteData.details;
    const requestedBaseQuoteNumber = baseQuoteNumber_(revisionOf);
    // 수정본은 원본을 조회했을 때의 부서(revisionDepartment)를 그대로 사용해야 한다.
    // 폼의 작성자 드롭다운에서 다른 사람을 선택했더라도, 폴더/대장 조회는 반드시 원본이 저장된
    // 부서를 기준으로 해야 로컬 에이전트가 같은 폴더에 파일을 저장할 수 있다.
    if (requestedBaseQuoteNumber && String(revisionDepartment || '').trim()) {
      details.authorDepartment = String(revisionDepartment).trim();
    }
    const rootFolderId = resolveRootFolderId(details);

    // 수정본은 원본을 조회했던 연도(revisionYear)를 그대로 사용해야 한다.
    // 견적일자를 오늘 날짜로 채워도 원본이 저장된 연도의 대장을 찾도록 한다.
    // 미래 연도·존재하지 않는 연도가 들어오면 대장/폴더를 새로 만들지 않고 즉시 실패시킨다.
    const nowYear = new Date().getFullYear();
    let currentYear;
    let system;
    const parsedRevisionYear = Number(revisionYear);
    if (requestedBaseQuoteNumber) {
      if (Number.isInteger(parsedRevisionYear) && parsedRevisionYear >= 2000 && parsedRevisionYear <= nowYear) {
        currentYear = parsedRevisionYear;
      } else {
        const dateMatches = String(details.quoteDate || '').match(/(\d{4})/);
        currentYear = dateMatches ? Math.min(parseInt(dateMatches[1], 10), nowYear) : nowYear;
      }
      // 수정 대상은 기존 연도 폴더/대장이 반드시 있어야 한다 — 없으면 새로 만들지 않고 실패 응답한다.
      system = getExistingYearSystem_(currentYear, rootFolderId);
      if (!system || findLedgerRow_(system.ledgerSheet, requestedBaseQuoteNumber) < 2) {
        lock.releaseLock();
        return { success: false, message: '수정할 원본 견적을 찾을 수 없습니다. 연도를 확인해 주세요.' };
      }
    } else {
      const dateMatches = String(details.quoteDate || '').match(/(\d{4})/);
      currentYear = dateMatches ? parseInt(dateMatches[1], 10) : nowYear;
      system = getYearSystem(currentYear, rootFolderId);
    }

    let realQuoteNumber;
    let baseQuoteNumber;
    let revisionNumber = 0;
    if (requestedBaseQuoteNumber) {
      baseQuoteNumber = requestedBaseQuoteNumber;
      revisionNumber = getNextRevisionNumber_(system.ledger, system.ledgerSheet, baseQuoteNumber);
      realQuoteNumber = `${baseQuoteNumber}_Rev${String(revisionNumber).padStart(2, '0')}`;
      details.baseQuoteNumber = baseQuoteNumber;
      details.revisionNumber = revisionNumber;
      details.folderClientName = getQuoteFolderClientName_(system.ledger, baseQuoteNumber) || details.clientName;
    } else {
      realQuoteNumber = getNextQuoteNumber(rootFolderId, details.authorDepartment);
      baseQuoteNumber = realQuoteNumber;
      details.folderClientName = details.clientName;
    }
    details.quoteNumber = realQuoteNumber;
    // 로컬 에이전트는 details.quoteDate(항상 오늘 날짜)가 아니라 이 값으로 저장 연도 폴더를 결정해야
    // 수정본을 원본과 같은 연도 폴더에 저장할 수 있다.
    details.year = currentYear;

    const items = quoteData.items || [];
    const itemParts = splitQuoteItems_(items);
    const finalVatPrice = Math.round(items.reduce(function(sum, item) {
      return sum + Number(item.totalPrice || 0);
    }, 0) * 1.1);

    // 파일 생성 작업을 먼저 대기열에 등록한다.
    // 대장 행은 에이전트가 XLSX/PDF 생성에 성공했다는 완료 보고를 보낸 뒤 processAgentResults에서 기록한다.
    enqueueLocalSave_(details, items, finalVatPrice, createDraft, customSubject);

    let returnMessage = '[접수] 견적번호: ' + realQuoteNumber;
    if (itemParts.length > 1) {
      returnMessage += `\n${itemParts.length}개 파일로 분할 생성됩니다. (파일당 최대 ${ITEMS_PER_FILE}개 품목)`;
    }
    returnMessage += '\n파일 생성이 성공한 뒤 견적관리대장에 자동 기록됩니다.';

    if (createDraft) {
      if (details.clientEmail && details.clientEmail.indexOf('@') > -1) {
        // Gmail 초안 첨부용으로만 임시 PDF를 만들고, 초안 생성 직후 구글 드라이브에서 삭제한다.
        const pdfBlobs = itemParts.map(function(partItems, partIndex) {
          const partDetails = Object.assign({}, details, {
            quoteNumber: quotePartNumber_(realQuoteNumber, partIndex, itemParts.length)
          });
          return createTempQuotePdf_({ details: partDetails, items: partItems }, system, partDetails);
        });
        GmailApp.createDraft(
          details.clientEmail,
          customSubject || `[CIMON] ${details.clientName} - 제품 견적서 송부 드립니다.`,
          customBody || makeDefaultEmailBody(details),
          {
            attachments: pdfBlobs,
            name: details.authorName || 'CIMON'
          }
        );
        returnMessage += '\n\nGmail 초안 생성 완료!';
      } else {
        returnMessage += '\n\n이메일 주소 오류: Gmail 초안 생성 실패';
      }
    }

    lock.releaseLock();
    return {
      success: true,
      url: '',
      pdfUrl: '',
      folderUrl: system.folder.getUrl(),
      sheetUrl: '',
      newQuoteNumber: realQuoteNumber,
      baseQuoteNumber: baseQuoteNumber,
      revisionNumber: revisionNumber || undefined,
      message: returnMessage
    };
  } catch (err) {
    lock.releaseLock();
    const errorText = String(err && err.toString ? err.toString() : err);
    if (/DriveApp|액세스가 거부|Access denied|권한이 없습니다/i.test(errorText)) {
      return {
        success: false,
        message: '견적 저장에 필요한 Google Drive 권한이 없습니다. 관리자에게 부서 폴더·대장 템플릿·견적에이전트 폴더의 편집 권한을 요청해 주세요. 관리자 계정에서 diagnoseUserDriveAccess를 실행하면 누락된 권한을 확인할 수 있습니다.'
      };
    }
    return { success: false, message: '작업 실패: ' + errorText };
  }
}

// Gmail 초안 첨부용 임시 PDF 생성. 시트 복사 → 채우기 → PDF 내보내기 → 원본 시트 즉시 삭제.
function createTempQuotePdf_(quoteData, system, details) {
  const tempName = `임시_${sanitizeDriveName(details.quoteNumber)}_${Date.now()}`;
  const templateFile = DriveApp.getFileById(CONFIG.TEMPLATE_ID);
  const newFile = templateFile.makeCopy(tempName, system.folder);
  try {
    const spreadsheet = SpreadsheetApp.open(newFile);
    fillSheetData(spreadsheet, quoteData);
    SpreadsheetApp.flush();
    const sheetId = spreadsheet.getSheets()[0].getSheetId();
    return createPdfBlob(newFile.getId(), sheetId, `${sanitizeDriveName(details.quoteNumber)}_${sanitizeDriveName(details.clientName)}_견적서`);
  } finally {
    permanentlyDeleteFile_(newFile.getId());
  }
}

// 드라이브 파일을 휴지통이 아닌 영구 삭제로 제거한다 (용량 즉시 회수).
// 공유 드라이브(Shared Drive) 항목 삭제에는 supportsAllDrives 파라미터가 필요하다.
function permanentlyDeleteFile_(fileId) {
  try {
    const res = UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/files/' + fileId + '?supportsAllDrives=true', {
      method: 'delete',
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      muteHttpExceptions: true
    });
    if (res.getResponseCode() !== 204) {
      Logger.log('파일 영구 삭제 실패(' + fileId + '): ' + res.getContentText());
    }
  } catch (err) {
    Logger.log('파일 영구 삭제 예외(' + fileId + '): ' + err.toString());
  }
}

// ── 사내 PC 로컬 저장 에이전트 (Google Drive 폴더 연동) 끝 ────────────────

function fillSheetData(spreadsheet, quoteData) {
  const sheet = spreadsheet.getSheets()[0];
  const details = quoteData.details || {};
  const items = quoteData.items || [];

  setValueIfRangeExists(sheet, CELL_MAP.clientName, details.clientName);
  setValueIfRangeExists(sheet, CELL_MAP.clientContactPerson, details.clientContactPerson);
  setValueIfRangeExists(sheet, CELL_MAP.clientPhone, details.clientPhone);
  setValueIfRangeExists(sheet, CELL_MAP.clientEmail, details.clientEmail);
  setValueIfRangeExists(sheet, CELL_MAP.quoteNumber, details.quoteNumber);
  setValueIfRangeExists(sheet, CELL_MAP.quoteDate, details.quoteDate);
  setValueIfRangeExists(sheet, CELL_MAP.deliveryLocation, details.deliveryLocation);
  setValueIfRangeExists(sheet, CELL_MAP.deliveryDeadline, details.deliveryDeadline);
  setValueIfRangeExists(sheet, CELL_MAP.paymentTerms, details.paymentTerms);
  setValueIfRangeExists(sheet, CELL_MAP.validityPeriod, details.validityPeriod);
  setValueIfRangeExists(sheet, CELL_MAP.packing, details.packing);
  setValueIfRangeExists(sheet, CELL_MAP.authorName, details.authorName);
  setValueIfRangeExists(sheet, CELL_MAP.authorPhone, details.authorPhone);
  setValueIfRangeExists(sheet, CELL_MAP.authorEmail, details.authorEmail);
  setValueIfRangeExists(sheet, CELL_MAP.notes, details.notes || '');

  try {
    sheet.getRange('D16:D100').setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);
  } catch (err) {
    // Ignore wrapping failures on templates with different ranges.
  }

  let supplyTotal = 0;
  for (let i = 0; i < items.length; i++) {
    const row = CELL_MAP.itemStartRow + i;
    const item = items[i];
    const quantity = Number(item.quantity || 0);
    const unitPrice = Number(item.unitPrice || 0);
    const multiplierValue = Number(item.multiplier || 1);
    const multiplier = isFinite(multiplierValue) && multiplierValue > 0 ? multiplierValue : 1;
    const effectiveUnitPrice = unitPrice * multiplier;
    const totalPrice = Math.round(quantity * effectiveUnitPrice);

    setValueIfRangeExists(sheet, `A${row}`, i + 1);
    setValueIfRangeExists(sheet, `B${row}`, item.name || '');
    setValueIfRangeExists(sheet, `D${row}`, item.spec || '');
    setValueIfRangeExists(sheet, `G${row}`, quantity);
    // 템플릿에는 배율 열이 없으므로 단가 칸에 배율을 반영한 유효 단가를 기록한다.
    setValueIfRangeExists(sheet, `H${row}`, effectiveUnitPrice);
    setValueIfRangeExists(sheet, `I${row}`, totalPrice);

    supplyTotal += totalPrice;
  }

  const vatIncludedPrice = Math.round(supplyTotal * 1.1);
  const formattedPrice = vatIncludedPrice.toLocaleString('ko-KR');
  setValueIfRangeExists(sheet, CELL_MAP.totalPriceCell, `총 견적 금액(VAT포함): ${formattedPrice} 원`);

  return vatIncludedPrice;
}

function createPdfBlob(spreadsheetId, sheetId, fileName) {
  const url =
    'https://docs.google.com/spreadsheets/d/' +
    spreadsheetId +
    '/export?format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=' +
    sheetId;

  const params = {
    method: 'GET',
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
    muteHttpExceptions: true
  };

  return UrlFetchApp.fetch(url, params).getBlob().setName(fileName + '.pdf');
}

function saveToLedgerSheet(sheet, details, items, totalPrice, fileUrl) {
  // 신규/수정 행을 고정 열 번호(LEDGER_COL)에 그대로 쓰기 전에 실제 헤더가 예상과 일치하는지
  // 검증한다 — 대장 템플릿의 열 구성이 다르면 조용히 잘못된 열에 쓰는 대신 즉시 실패시킨다.
  assertLedgerLayout_(sheet);
  const quoteDate = String(details.quoteDate || '');
  const dateMatches = quoteDate.match(/(\d{4}).*?(\d{1,2}).*?(\d{1,2})/);
  const year = dateMatches ? dateMatches[1] : new Date().getFullYear();
  const month = dateMatches ? dateMatches[2] : new Date().getMonth() + 1;
  const day = dateMatches ? dateMatches[3] : new Date().getDate();
  const firstItem = items[0] || {};
  const productName = items.length > 1 ? `${firstItem.name} 외 ${items.length - 1}건` : firstItem.name || '';
  const categoryRules = [
    ['SCADA PRO', /SCADA\s*PRO/i],
    ['SCADA', /SCADA/i],
    ['TOUCH MONITOR', /TOUCH\s*MONITOR/i],
    ['BOX PC', /BOX\s*PC|\bNB\d/i],
    ['Hybird', /HYBRID|HYBIRD/i],
    ['Accessory', /ACCESSORY|ACCESSARY|액세서리/i],
    ['eXT', /\beXT\d*\b/i],
    ['XPANEL', /XPANEL/i],
    ['PLC', /\bPLC\b|\bCM[013]\b|NET\/RIO|CIMON-NET|REMOTE\s*IO|\bRIO\b/i],
    ['TOUCH', /TOUCH|50000_70000|5000SERIES|500SERIES|\biN[TP]\d/i]
  ];
  const productCategories = [];
  items.forEach(function(item) {
    const source = String(item.type || '') + ' ' + String(item.name || '');
    const matched = categoryRules.find(function(rule) { return rule[1].test(source); });
    if (matched && productCategories.indexOf(matched[0]) === -1) productCategories.push(matched[0]);
  });

  // 수정본은 원본 행을 갱신하고, 원본 견적번호는 대장에 그대로 유지한다.
  const ledgerQuoteNumber = String(details.baseQuoteNumber || details.quoteNumber || '').trim();
  // 대장 NO는 기본 견적번호의 일련번호(-NNN)와 항상 일치시킨다.
  // 견적번호 파싱에 실패한 경우에만 행 개수로 보정한다.
  const seqMatch = ledgerQuoteNumber.match(/-(\d+)\s*$/);
  const no = seqMatch ? parseInt(seqMatch[1], 10) : sheet.getLastRow();

  // F열(견적번호)부터 N열(비고)까지 9개 값 — E열(발주)은 사용자가 대장에서 직접
  // 관리하는 열이므로 신규/수정 저장 로직에서는 절대 건드리지 않는다.
  const quoteInfoValues = [
    ledgerQuoteNumber,
    details.clientName,
    details.clientContactPerson,
    details.clientPhone,
    details.clientEmail,
    productCategories.join(', '),
    productName,
    totalPrice,
    details.notes || ''
  ];

  // 신규 견적도 완료 보고 재처리 시 같은 견적번호 행을 중복 append하지 않도록 먼저 찾는다.
  const existingRow = findLedgerRow_(sheet, ledgerQuoteNumber);
  if (details.baseQuoteNumber && existingRow < 2) {
    throw new Error('수정 대상 원본 행을 찾을 수 없습니다.');
  }

  // 스냅샷 저장을 먼저 시도한다 — 스냅샷이 실패하면(예: 셀 글자수 한도) 대장 행은 아직
  // 손대지 않은 상태이므로 "일부만 반영된" 상태가 되지 않는다.
  saveQuoteSnapshot_(sheet.getParent(), details, items, totalPrice);

  if (existingRow >= 2) {
    // 수정본: A~D열(날짜)과 E열(발주)은 원본 값을 그대로 두고 F~N열만 갱신한다.
    // O열(파일링크)은 updateLedgerLink_이 원본·Rev 링크를 누적하므로 여기서 건드리지 않는다.
    sheet.getRange(existingRow, LEDGER_COL.QUOTE_NUMBER, 1, quoteInfoValues.length).setValues([quoteInfoValues]);
  } else {
    // 신규 견적: A~D, E(발주 공란), F~N, O(파일링크)를 한 번에 추가한다.
    sheet.appendRow([no, year, month, day, ''].concat(quoteInfoValues).concat([fileUrl]));
  }
}

function generateAIContent(prompt) {
  if (!CONFIG.GEMINI_API_KEY) {
    return 'Gemini API key가 설정되지 않았습니다.';
  }

  const url = CONFIG.GEMINI_URL_BASE + CONFIG.GEMINI_API_KEY;
  const payload = { contents: [{ parts: [{ text: prompt }] }] };

  try {
    const response = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });
    const result = JSON.parse(response.getContentText());
    return result.candidates && result.candidates[0] && result.candidates[0].content && result.candidates[0].content.parts
      ? result.candidates[0].content.parts[0].text.trim()
      : '응답 없음';
  } catch (err) {
    return '에러';
  }
}

// ── 파일링크 생성 헬퍼 (공용) ───────────────────────────────────────────────
// 견적 번호/업체명 → 대장 O열(파일링크)에 기록할 서명된 링크 목록을 만든다.

function signedFileUrl_(dept, year, folderName, fileName) {
  const publicBaseUrl = 'http://172.35.12.36:8790';
  const fileLinkSecret = 'CIMON-FILES-2026-jR4wN8sQ2tVx7LmE';
  const relPath = [dept, String(year), folderName, fileName].map(encodeURIComponent).join('/');
  // 한글(멀티바이트)이 들어간 경로를 그대로 서명하면 Apps Script와 Node가 문자열→바이트 처리가
  // 달라 서명이 어긋난다. URL 인코딩된 ASCII 경로로 서명하면 양쪽 HMAC이 정확히 일치한다.
  const sigBytes = Utilities.computeHmacSha256Signature(relPath, fileLinkSecret);
  const sig = sigBytes.map(function(b) {
    const v = (b < 0 ? b + 256 : b).toString(16);
    return v.length === 1 ? '0' + v : v;
  }).join('');
  return `${publicBaseUrl}/files/${relPath}?k=${sig}`;
}

function buildRowLinkItems_(dept, year, quoteNumber, company, snapshotRows) {
  const baseQuote = baseQuoteNumber_(quoteNumber);
  const matchedSnapshots = snapshotRows.filter(function(row) {
    return String(row[1] || '').trim() === baseQuote;
  });

  // 실제 폴더명에 사용된 업체명(스냅샷의 folderClientName)을 우선 사용한다.
  let folderClientName = company;
  for (let si = matchedSnapshots.length - 1; si >= 0; si--) {
    try {
      const stored = JSON.parse(String(matchedSnapshots[si][4] || '{}'));
      const storedClient = String((stored.details && (stored.details.folderClientName || stored.details.clientName)) || '').trim();
      if (storedClient) {
        folderClientName = storedClient;
        break;
      }
    } catch (err) { /* 다음 스냅샷 행으로 넘어간다 */ }
  }

  const folderName = `${baseQuote}_${folderClientName}`;
  const linkItems = [];
  const seenUrls = {};

  const origFileName = `${baseQuote}_${folderClientName}_견적서.pdf`;
  const origUrl = signedFileUrl_(dept, year, folderName, origFileName);
  linkItems.push({ text: `원본: ${origFileName}`, url: origUrl });
  seenUrls[origUrl] = true;

  matchedSnapshots.forEach(function(row) {
    const outNum = String(row[0] || '').trim();
    const revMatch = outNum.match(/_Rev(\d+)$/i);
    if (!revMatch) return;
    const revNum = parseInt(revMatch[1], 10);
    const revFileName = `${outNum}_${folderClientName}_견적서.pdf`;
    const revUrl = signedFileUrl_(dept, year, folderName, revFileName);
    if (seenUrls[revUrl]) return;
    seenUrls[revUrl] = true;
    linkItems.push({ text: `Rev${String(revNum).padStart(2, '0')}: ${revFileName}`, url: revUrl });
  });

  linkItems.sort(function(a, b) {
    const aRev = a.text.match(/^Rev(\d+)/);
    const bRev = b.text.match(/^Rev(\d+)/);
    const aNum = aRev ? parseInt(aRev[1], 10) : 0;
    const bNum = bRev ? parseInt(bRev[1], 10) : 0;
    return aNum - bNum;
  });
  return linkItems;
}

function writeLinkItemsToCell_(sheet, rowIndex, linkItems) {
  if (!linkItems || linkItems.length === 0) return;
  const separator = ' | ';
  const combinedText = linkItems.map(function(item) { return item.text; }).join(separator);
  const builder = SpreadsheetApp.newRichTextValue().setText(combinedText);
  let offset = 0;
  linkItems.forEach(function(item) {
    if (item.url) builder.setLinkUrl(offset, offset + item.text.length, item.url);
    offset += item.text.length + separator.length;
  });
  sheet.getRange(rowIndex, LEDGER_COL.FILE_LINK).setRichTextValue(builder.build());
}

// 견적 목록 조회 시 누락된 파일링크를 "그 자리에서" 자동 복구한다.
// 반환할 rows 배열에도 복구된 링크를 반영해서 화면에 즉시 보이게 한다.
function repairEmptyLedgerLinks_(system, dept, year, rows) {
  const sheet = system.ledgerSheet;
  const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
  const snapshotRows = (snapshot && snapshot.getLastRow() >= 2)
    ? snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues()
    : [];
  const linkCol = LEDGER_COL.FILE_LINK - 1;
  const quoteCol = LEDGER_COL.QUOTE_NUMBER - 1;
  const companyCol = LEDGER_COL.CLIENT_NAME - 1;
  let repaired = 0;
  for (let i = 0; i < rows.length; i++) {
    const row = rows[i];
    if (row.links[linkCol]) continue; // 이미 링크가 있으면 건드리지 않는다.
    const quoteNumber = String(row.values[quoteCol] || '').trim();
    const company = String(row.values[companyCol] || '').trim();
    if (!quoteNumber || !company) continue;
    const linkItems = buildRowLinkItems_(dept, year, quoteNumber, company, snapshotRows);
    if (!linkItems.length) continue;
    writeLinkItemsToCell_(sheet, i + 2, linkItems);
    row.links[linkCol] = linkItems[0].url; // 화면에 즉시 표시되도록 반환 데이터에 반영
    repaired++;
  }
  if (repaired > 0) SpreadsheetApp.flush();
  return repaired;
}

// ── 파일링크 복구 유틸리티 ──────────────────────────────────────────────────
// 에이전트 큐 삭제나 동기화 지연 등으로 대장 파일링크가 누락된 모든 견적 행의 링크를
// 스냅샷 기록을 바탕으로 자동 계산하여 대장에 복구한다.
// 사용 방법: 편집기 상단 함수 드롭다운에서 autoRepairAllMissingLedgerLinks 선택 → [실행] 클릭
function autoRepairAllMissingLedgerLinks() {
  const currentYear = new Date().getFullYear();
  for (const dept of Object.keys(CONFIG.DEPARTMENT_ROOTS)) {
    const rootFolderId = CONFIG.DEPARTMENT_ROOTS[dept];
    const system = getExistingYearSystem_(currentYear, rootFolderId);
    if (!system) continue;

    const sheet = system.ledgerSheet;
    const lastRow = sheet.getLastRow();
    if (lastRow < 2) continue;

    const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
    const snapshotRows = (snapshot && snapshot.getLastRow() >= 2)
      ? snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues()
      : [];

    let repaired = 0;
    for (let r = 2; r <= lastRow; r++) {
      const quoteNumber = String(sheet.getRange(r, LEDGER_COL.QUOTE_NUMBER).getDisplayValue() || '').trim();
      const company = String(sheet.getRange(r, LEDGER_COL.CLIENT_NAME).getDisplayValue() || '').trim();
      if (!quoteNumber || !company) continue;
      const linkItems = buildRowLinkItems_(dept, currentYear, quoteNumber, company, snapshotRows);
      if (!linkItems.length) continue;
      writeLinkItemsToCell_(sheet, r, linkItems);
      repaired++;
    }
    SpreadsheetApp.flush();
    Logger.log(`[부서: ${dept}] 파일링크 복구 완료: ${repaired}행`);
  }
  Logger.log('전체 부서 파일링크 복구 작업 완료!');
}

function repairMissingLedgerLink() {
  autoRepairAllMissingLedgerLinks();
}

      Logger.log(`[부서: ${dept}] 견적관리대장이 이미 비어 있습니다.`);
    }

    // 1-2. 스냅샷 시트(견적원본데이터) 행 비우기 (2행부터 마지막 행까지 삭제)
    const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
    if (snapshot) {
      const snapLastRow = snapshot.getLastRow();
      if (snapLastRow >= 2) {
        snapshot.deleteRows(2, snapLastRow - 1);
        Logger.log(`[부서: ${dept}] 견적원본데이터(스냅샷) ${snapLastRow - 1}개 행 삭제 완료`);

      }

      try {
        const subFolder = getAgentSubfolder_(subName);
        const files = subFolder.getFiles();

        while (files.hasNext()) {
          const file = files.next();
          permanentlyDeleteFile_(file.getId());
          deletedCount++;
        }
        Logger.log(`[큐 폴더: ${subName}] 잔여 파일 ${deletedCount}개 정리 완료`);
      } catch (err) {
        Logger.log(`[큐 폴더: ${subName}] 정리 실패: ${err.toString()}`);
      }
    });
  } catch (agentErr) {
    Logger.log(`에이전트 큐 정리 오류: ${agentErr.toString()}`);
  }

  Logger.log('====================================================');
  Logger.log('전체 부서(기술영업/영업/프로젝트) 초기화가 완료되었습니다!');
  Logger.log('이제 각 부서의 첫 견적서는 001번부터 정상 채번됩니다.');
  Logger.log('====================================================');
}

  // URL 인코딩된 ASCII 경로로 서명한 값 (에이전트가 검증하는 방식과 동일)
  const relPath = ['기술영업', '2026', '기술영업 2609-001_test1', '기술영업 2609-001_test1_견적서.pdf']
    .map(encodeURIComponent).join('/');
  const bytes = Utilities.computeHmacSha256Signature(relPath, secret);
  const hex = bytes.map(function(b) {
    const v = (b < 0 ? b + 256 : b).toString(16);
    return v.length === 1 ? '0' + v : v;
  }).join('');
  Logger.log('Apps Script HMAC(인코딩 경로, hex): ' + hex);
  Logger.log('이 값이 에이전트가 URL로 검증하는 서명과 일치해야 합니다.');
}

// ── 대장만 있고 로컬 파일이 없는 견적 재처리 유틸리티 ────────────────────────
// Apps Script가 대장은 저장했지만 pending 큐 기록/Drive 동기화가 실패한 경우,
// 견적원본데이터 스냅샷에서 원본/리비전 작업을 다시 만들어 로컬 에이전트에 전달한다.
// 실행 방법: 함수 선택 → requeueMissingLocalQuoteJobs → 실행
function requeueMissingLocalQuoteJobs() {
  let queued = 0;
  const currentYear = new Date().getFullYear();

  for (const dept of Object.keys(CONFIG.DEPARTMENT_ROOTS)) {
    const rootFolderId = CONFIG.DEPARTMENT_ROOTS[dept];
    const years = getAvailableLedgerYears_(rootFolderId, currentYear);

    for (const year of years) {
      const system = getExistingYearSystem_(year, rootFolderId);
      if (!system) continue;

      const sheet = system.ledgerSheet;
      const lastRow = sheet.getLastRow();
      if (lastRow < 2) continue;

      const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
      if (!snapshot || snapshot.getLastRow() < 2) continue;
      const snapshotRows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues();

      for (let rowIndex = 2; rowIndex <= lastRow; rowIndex++) {
        const quoteNumber = String(sheet.getRange(rowIndex, LEDGER_COL.QUOTE_NUMBER).getDisplayValue() || '').trim();
        const fileLink = String(sheet.getRange(rowIndex, LEDGER_COL.FILE_LINK).getDisplayValue() || '').trim();
        if (!quoteNumber || fileLink) continue;

        const baseQuote = baseQuoteNumber_(quoteNumber);
        const matchingSnapshots = snapshotRows.filter(function(row) {
          return String(row[1] || '').trim() === baseQuote;
        });
        if (matchingSnapshots.length === 0) {
          Logger.log(`[재처리 건너뜀] ${dept}/${year}/${baseQuote}: 원본 스냅샷 없음`);
          continue;
        }

        for (const snapshotRow of matchingSnapshots) {
          let stored;
          try {
            stored = JSON.parse(String(snapshotRow[4] || '{}'));
          } catch (err) {
            Logger.log(`[재처리 실패] ${baseQuote}: 스냅샷 JSON 해석 실패`);
            continue;
          }
          const details = stored.details || {};
          const outputNumber = String(snapshotRow[0] || details.quoteNumber || '').trim();
          if (!outputNumber) continue;

          details.authorDepartment = dept;
          details.year = year;
          details.quoteNumber = outputNumber;
          if (baseQuote !== outputNumber && !details.baseQuoteNumber) details.baseQuoteNumber = baseQuote;

          const pending = getAgentSubfolder_('pending');
          const pendingName = `${sanitizeDriveName(outputNumber)}.json`;
          const existing = pending.getFilesByName(pendingName);
          if (existing.hasNext()) continue;

          enqueueLocalSave_(details, stored.items || [], Number(stored.totalPrice || 0), false, '');
          queued++;
          Logger.log(`[재처리 큐 등록] ${dept}/${year}/${outputNumber}`);
        }
      }
    }
  }

  Logger.log(`대장 누락 파일 재처리 큐 등록 완료: ${queued}건`);
}

// ── 사용자별 Drive 접근 권한 진단 ────────────────────────────────────────────
// 실행한 "현재 로그인 계정" 기준으로, 시스템이 사용하는 공유 Drive 리소스에
// 접근 가능한지 항목별로 확인한다. Apps Script 편집기에서 선택 → 실행 → 로그 확인.
// 다른 작성자 계정이 "액세스가 거부됨: DriveApp" 오류를 겪을 때 원인 파악에 사용한다.
function diagnoseUserDriveAccess() {
  const email = String(Session.getActiveUser().getEmail() || '').trim();
  Logger.log('=== 현재 실행 계정: ' + email + ' ===');
  Logger.log('아래 항목이 모두 OK이어야 이 계정으로 견적 폴더 파일 목록 조회가 가능합니다.');
  Logger.log('');

  function check(label, fn) {
    try {
      const result = fn();
      Logger.log('OK   ' + label + (result ? ' — ' + result : ''));
    } catch (err) {
      Logger.log('FAIL ' + label + ' — ' + err.toString());
    }
  }

  check('작성자 DB 시트', function() {
    const ss = SpreadsheetApp.openById(CONFIG.AUTHOR_DB_SHEET_ID);
    return ss.getName();
  });

  for (const dept of Object.keys(CONFIG.DEPARTMENT_ROOTS)) {
    check('부서 폴더 [' + dept + ']', function() {
      return DriveApp.getFolderById(CONFIG.DEPARTMENT_ROOTS[dept]).getName();
    });
  }

  check('견적에이전트 폴더', function() {
    return getAgentFolder_().getName();
  });

  check('견적에이전트/문서 폴더', function() {
    return getAgentSubfolder_('문서').getName();
  });

  check('견적서 템플릿', function() {
    return DriveApp.getFileById(CONFIG.TEMPLATE_ID).getName();
  });

  check('대장 템플릿', function() {
    return DriveApp.getFileById(CONFIG.LEDGER_TEMPLATE_ID).getName();
  });

  Logger.log('');
  Logger.log('FAIL이 있는 항목은 해당 계정에 공유(또는 편집) 권한이 없는 것입니다.');
  Logger.log('관리자에게 해당 리소스를 이 계정(또는 작성자 그룹)과 공유해 달라고 요청하세요.');
}
