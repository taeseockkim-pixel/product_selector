// 대장 시트가 운영 열 구성(E=발주, F=작성자, G=견적번호, P=파일링크)을 갖는지 검증한다.
function assertLedgerLayout_(ledgerSheet) {
  const lastColumn = Math.max(ledgerSheet.getLastColumn(), LEDGER_COL.FILE_LINK);
  const headers = ledgerSheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0];
  const orderHeader = String(headers[LEDGER_COL.ORDER - 1] || '');
  const authorHeader = String(headers[LEDGER_COL.AUTHOR - 1] || '');
  const quoteNumberHeader = String(headers[LEDGER_COL.QUOTE_NUMBER - 1] || '');
  const fileLinkHeader = String(headers[LEDGER_COL.FILE_LINK - 1] || '');
  if (
    orderHeader.indexOf('발주') === -1 ||
    authorHeader.indexOf('작성자') === -1 ||
    quoteNumberHeader.indexOf('견적번호') === -1 ||
    fileLinkHeader.indexOf('파일링크') === -1
  ) {
    throw new Error(
      '견적관리대장의 열 구성이 예상과 다릅니다(E열=발주, F열=작성자, G열=견적번호, P열=파일링크이어야 합니다). ' +
      'LEDGER_TEMPLATE_ID 대장 템플릿 또는 해당 연도 시트의 A:Q 헤더를 확인해 주세요.'
    );
  }
  return ledgerSheet;
}

function getQuoteLedgerFromReact(requestedYear, requestedDepartment) {
  try {
    const t0 = Date.now();
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return {
        success: false,
        message: access.message || '접속 권한이 없습니다.',
        email: access.email || ''
      };
    }

    // 관리자는 모든 부서의 대장을 선택해 조회할 수 있다.
    // 일반 사용자는 본인 등록 부서의 대장만 조회할 수 있다.
    let department = access.author.department;
    const targetDept = String(requestedDepartment || '').trim();
    if (access.isAdmin && targetDept && CONFIG.DEPARTMENT_ROOTS[targetDept]) {
      department = targetDept;
    }

    // 인증된 다음에만 완료 보고를 처리하고, 일반 계정은 자기 부서의 결과만 처리한다.
    // 관리자는 목록에서 선택한 부서만 즉시 반영한다(주기 트리거는 기존대로 전체 처리).
    try {
      processAgentResults(2, department);
    } catch (agentErr) {
      Logger.log('대장 조회 중 에이전트 완료 보고 반영 예외: ' + agentErr.toString());
    }

    const rootFolderId = resolveRootFolderId({ authorDepartment: department });
    const currentYear = new Date().getFullYear();
    const parsedYear = Number(requestedYear);
    const year = Number.isInteger(parsedYear) && parsedYear >= 2000 && parsedYear <= currentYear
      ? parsedYear
      : currentYear;
    const availableYears = getAvailableLedgerYears_(rootFolderId, currentYear);
    const system = getExistingYearSystem_(year, rootFolderId);
    const baseResult = {
      success: true,
      availableYears: availableYears,
      department: department,
      isAdmin: Boolean(access.isAdmin),
      availableDepartments: access.isAdmin ? Object.keys(CONFIG.DEPARTMENT_ROOTS) : [department]
    };
    if (!system) {
      return Object.assign({ headers: [], rows: [] }, baseResult);
    }
    const sheet = system.ledgerSheet;
    const lastRow = sheet.getLastRow();
    const lastColumn = sheet.getLastColumn();

    if (lastRow < 1 || lastColumn < 1) {
      return Object.assign({ headers: [], rows: [] }, baseResult);
    }

    const range = sheet.getRange(1, 1, lastRow, lastColumn);
    const displayValues = range.getDisplayValues();
    const headers = displayValues[0];

    // P열(파일링크)에만 URL 링크가 걸려 있으므로, 전체 시트가 아닌 P열만 1회성으로 읽는다.
    const fileLinkCol = LEDGER_COL.FILE_LINK;
    const richTextLinks = (lastRow >= 2 && lastColumn >= fileLinkCol)
      ? sheet.getRange(2, fileLinkCol, lastRow - 1, 1).getRichTextValues()
      : [];

    // 대장에 '적용현장' 열이 없으면 대장 시트의 마지막 열에 '적용현장' 헤더를 자동 추가한다.
    let siteCol = headers.findIndex(function(h) {
      return /적용\s*현장|현장명|^현장$/.test(String(h || '').trim());
    });
    if (siteCol < 0 && lastColumn >= 1) {
      try {
        sheet.getRange(1, lastColumn + 1).setValue('적용현장');
        headers.push('적용현장');
        siteCol = headers.length - 1;
        SpreadsheetApp.flush();
      } catch (colErr) {
        headers.push('적용현장');
      }
    }

    // 각 견적번호별 작성자 이름 맵 구축 (대장 검색에서 작성자 이름 검색 지원용)
    const authorMap = {};
    const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
    if (snapshot && snapshot.getLastRow() >= 2) {
      const snapRows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues();
      snapRows.forEach(function(sRow) {
        const q0 = String(sRow[0] || '').trim();
        const q1 = String(sRow[1] || '').trim();
        let name = '';
        try {
          const parsed = JSON.parse(String(sRow[4] || '{}'));
          name = String((parsed.details && parsed.details.authorName) || '').trim();
        } catch (e) { /* noop */ }
        if (!name && sRow[3]) {
          name = String(sRow[3]).split('@')[0];
        }
        if (name) {
          if (q0) authorMap[q0] = name;
          if (q1) authorMap[q1] = name;
        }
      });
    }

    // 작성자 DB의 이메일->이름 맵으로 보정
    try {
      const authorRows = readAuthorRows_();
      const emailToName = {};
      authorRows.forEach(function(ar) {
        if (ar.email && ar.name) emailToName[String(ar.email).toLowerCase()] = ar.name;
      });
      Object.keys(authorMap).forEach(function(k) {
        const cur = authorMap[k];
        if (cur.indexOf('@') !== -1 && emailToName[cur.toLowerCase()]) {
          authorMap[k] = emailToName[cur.toLowerCase()];
        }
      });
    } catch (e) { /* noop */ }

    // 견적번호 열의 폰트 취소선을 읽어 라인삭제(struck) 상태를 프론트에 전달한다.
    const quoteCol = LEDGER_COL.QUOTE_NUMBER;
    const fontLines = lastRow >= 2 ? sheet.getRange(2, quoteCol, lastRow - 1, 1).getFontLines() : [];
    const fileLinkIndex = fileLinkCol - 1;

    const rows = displayValues.slice(1).map(function(values, rowIndex) {
      const paddedValues = values.slice();
      while (paddedValues.length < headers.length) {
        paddedValues.push('');
      }

      // P열(파일링크)의 링크만 richTextLinks에서 추출하고 나머지는 URL 문자열 체크로 가볍게 처리
      const linkRt = (richTextLinks[rowIndex] && richTextLinks[rowIndex][0]);
      let fileUrl = null;
      if (linkRt) {
        fileUrl = linkRt.getLinkUrl();
        if (!fileUrl) {
          const runs = linkRt.getRuns();
          for (let ri = 0; ri < runs.length; ri++) {
            const ru = runs[ri].getLinkUrl();
            if (ru) { fileUrl = ru; break; }
          }
        }
      }

      const links = paddedValues.map(function(val, columnIndex) {
        if (columnIndex === fileLinkIndex) {
          return fileUrl || (/^https?:\/\//i.test(val) ? val : null);
        }
        return /^https?:\/\//i.test(val) ? val : null;
      });

      const quoteNum = String(paddedValues[quoteCol - 1] || '').trim();
      const baseNum = quoteNum.replace(/_Rev\d+$/i, '').trim();
      const authorName = authorMap[quoteNum] || authorMap[baseNum] || String(paddedValues[LEDGER_COL.AUTHOR - 1] || '').trim();

      return {
        values: paddedValues,
        links: links,
        struck: fontLines[rowIndex] && fontLines[rowIndex][0] === 'line-through',
        authorName: authorName
      };
    });

    // 파일 생성이 완료 보고 반영보다 늦더라도, 목록 조회 시점에 누락된 파일링크를 즉시 복구한다.
    // (비동기 완료 보고를 기다리지 않고, 매 조회마다 빈 링크 행만 스냅샷 기반으로 다시 채운다.)
    repairEmptyLedgerLinks_(system, department, year, rows);

    Logger.log('[성능] getQuoteLedgerFromReact 완료: ' + (Date.now() - t0) + 'ms (행수: ' + rows.length + ')');
    return Object.assign({ headers: headers, rows: rows }, baseResult);
  } catch (err) {
    Logger.log('대장 조회 오류: ' + err.toString());
    return {
      success: false,
      message: '해당 부서 대장을 조회하지 못했습니다: ' + err.toString()
    };
  }
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    // Vercel 폴백 경로(주로 사용되지 않음) — Apps Script 프로젝트의 스크립트 속성(Script Properties)에
    // AGENT_TOKEN 값을 설정해야만 이 경로가 동작한다. 설정되어 있지 않으면 무조건 거부한다(fail-closed).
    // 설정 방법: Apps Script 편집기 > 프로젝트 설정 > 스크립트 속성 > AGENT_TOKEN 추가,
    // 그리고 Vercel 환경변수 APPS_SCRIPT_AGENT_TOKEN에 같은 값을 설정한다.
    const expectedToken = String(PropertiesService.getScriptProperties().getProperty('AGENT_TOKEN') || '');
    if (!expectedToken || String(payload.agentToken || '') !== expectedToken) {
      return jsonResponse_({ success: false, message: '토큰 인증에 실패했습니다.' });
    }
    const result = processQuoteFromReact(payload);
    return jsonResponse_(result);
  } catch (err) {
    return jsonResponse_({ success: false, message: err.toString() });
  }
}

function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ── 사내 PC 로컬 저장 에이전트 (Google Drive 폴더 연동) ───────────────────
// Apps Script는 대기 폴더에 견적 JSON 파일만 쓰고, 사내 PC의 에이전트가
// Google Drive 데스크톱 동기화로 파일을 받아 XLSX/PDF를 생성·저장한다.
// 에이전트는 완료 보고를 results 폴더에 쓰며, 1분 트리거가 대장 파일링크를 갱신한다.

// 최신 견적관리대장 사본(XLSX)을 에이전트 전달 폴더에 내보낸다.
// 에이전트가 견적 파일 생성 시 같은 연도 폴더에 함께 복사한다.
function exportLedgerToAgent_(system, year, quoteNumber) {
  try {
    const ledgerId = system.ledger.getId();
    const res = UrlFetchApp.fetch('https://docs.google.com/spreadsheets/d/' + ledgerId + '/export?format=xlsx', {
      method: 'get',
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      muteHttpExceptions: true
    });
    if (res.getResponseCode() !== 200) {
      Logger.log('대장 내보내기 실패: ' + res.getContentText());
      return;
    }
    const delivery = getAgentSubfolder_('delivery');
    const fileName = `대장_${sanitizeDriveName(quoteNumber)}.xlsx`;
    const blob = res.getBlob().setName(fileName);
    delivery.createFile(blob);
  } catch (err) {
    Logger.log('대장 사본 전달 오류: ' + err.toString());
  }
}

function getAgentFolder_() {
  const props = PropertiesService.getScriptProperties();
  const cachedId = props.getProperty('AGENT_FOLDER_ID');
  if (cachedId) {
    try {
      const folder = DriveApp.getFolderById(cachedId);
      if (folder && !folder.isTrashed()) return folder;
    } catch (err) { /* 폴더가 삭제되었거나 접근 불가 시 재탐색 */ }
  }
  const authorFile = DriveApp.getFileById(CONFIG.AUTHOR_DB_SHEET_ID);
  const parents = authorFile.getParents();
  const parent = parents.hasNext() ? parents.next() : DriveApp.getRootFolder();
  let folder = null;
  const iter = parent.getFoldersByName(CONFIG.AGENT_QUEUE_FOLDER_NAME);
  while (iter.hasNext()) {
    const candidate = iter.next();
    if (!candidate.isTrashed()) {
      folder = candidate;
      break;
    }
  }
  if (!folder) {
    folder = parent.createFolder(CONFIG.AGENT_QUEUE_FOLDER_NAME);
  }
  props.setProperty('AGENT_FOLDER_ID', folder.getId());
  return folder;
}

function getAgentSubfolder_(name) {
  const root = getAgentFolder_();
  const iter = root.getFoldersByName(name);
  while (iter.hasNext()) {
    const candidate = iter.next();
    if (!candidate.isTrashed()) return candidate;
  }
  return root.createFolder(name);
}

// 로컬 LOG 폴더(계정별/일자별)에 기록할 활동 로그를 Drive 동기화 폴더를 통해 에이전트로 전달한다.
// 브라우저는 HTTPS 페이지에서 사내 http 서버로 fetch할 수 없으므로 파일 큐를 경유한다.
function enqueueActivityLog_(account, eventType, detail) {
  try {
    const logs = getAgentSubfolder_('logs');
    const fileName = `log_${sanitizeDriveName(account || 'unknown')}_${Date.now()}.json`;
    logs.createFile(fileName, JSON.stringify({ account: account || 'unknown', eventType: eventType, detail: detail || '' }), MimeType.PLAIN_TEXT);
  } catch (err) {
    Logger.log('활동 로그 큐 작성 실패: ' + err.toString());
  }
}

function enqueueLocalSave_(details, items, vatTotal, createDraft, customSubject) {
  const pending = getAgentSubfolder_('pending');
  const fileName = `${sanitizeDriveName(details.quoteNumber)}.json`;
  const payload = JSON.stringify({
    details: details,
    items: items || [],
    vatTotal: vatTotal,
    createDraft: Boolean(createDraft),
    customSubject: customSubject || ''
  });
  pending.createFile(fileName, payload, MimeType.PLAIN_TEXT);
  ensureAgentTrigger_();
}

function splitQuoteItems_(items) {
  const parts = [];
  for (let i = 0; i < items.length; i += ITEMS_PER_FILE) {
    parts.push(items.slice(i, i + ITEMS_PER_FILE));
  }
  return parts.length > 0 ? parts : [[]];
}

function quotePartNumber_(baseQuoteNumber, partIndex, partCount) {
  return partCount > 1 ? `${baseQuoteNumber}_${partIndex + 1}` : baseQuoteNumber;
}

function ensureAgentTrigger_() {
  const exists = ScriptApp.getProjectTriggers().some(function(t) {
    return t.getHandlerFunction() === 'processAgentResults';
  });
  if (!exists) {
    ScriptApp.newTrigger('processAgentResults').timeBased().everyMinutes(1).create();
  }
}

// 에이전트가 XLSX/PDF 생성에 성공한 뒤에만 대장 행을 기록한다.
// 결과 JSON에 원본 상세 데이터가 포함되어 있어 재시도해도 동일한 행/스냅샷을 중복 생성하지 않는다.
function commitLedgerAfterAgentSuccess_(result) {
  if (!result || !result.details || !Array.isArray(result.items)) {
    throw new Error('완료 보고에 대장 저장용 견적 상세 데이터가 없습니다. 에이전트 최신 코드를 반영해 주세요.');
  }
  const rootFolderId = resolveRootFolderId({ authorDepartment: result.department });
  const year = Number(result.year) || new Date().getFullYear();
  const system = getExistingYearSystem_(year, rootFolderId);
  if (!system) throw new Error('대장을 찾을 수 없습니다(부서: ' + result.department + ', 연도: ' + year + ')');

  result.details.authorDepartment = result.department;
  result.details.year = year;
  result.details.quoteNumber = result.quoteNumber;
  if (result.ledgerQuoteNumber && result.revisionNumber > 0) {
    result.details.baseQuoteNumber = result.ledgerQuoteNumber;
    result.details.revisionNumber = result.revisionNumber;
  }
  saveToLedgerSheet(
    system.ledgerSheet,
    result.details,
    result.items,
    Number(result.vatTotal || result.details.totalPrice || 0),
    ''
  );
  return system;
}

// 1분마다 실행: 에이전트가 쓴 완료 보고를 반영해 대장 파일링크를 갱신하고 잔여 파일을 정리한다.
function processAgentResults(maxFiles, allowedDepartment) {
  const limit = Number.isInteger(maxFiles) && maxFiles > 0 ? maxFiles : 20;
  let results;
  try {
    results = getAgentSubfolder_('results');
  } catch (err) {
    return;
  }
  const files = results.getFiles();
  let scannedCount = 0;
  while (files.hasNext() && scannedCount < limit) {
    const file = files.next();
    scannedCount++;
    let result = null;
    try {
      result = JSON.parse(file.getBlob().getDataAsString('utf8'));
    } catch (err) {
      continue; // 아직 동기화 중인 파일 — 다음 트리거에서 재시도
    }

    if (
      allowedDepartment &&
      result &&
      normalizeAuthorDepartment_(result.department) !== normalizeAuthorDepartment_(allowedDepartment)
    ) {
      continue;
    }

    if (result && result.ok && result.department && result.quoteNumber && result.fileUrl) {
      const ledgerStartedAt = Date.now();
      try {
        // 파일 생성 성공을 먼저 확인한 뒤에만 대장 행을 만들고, 그 다음 링크를 기록한다.
        const system = commitLedgerAfterAgentSuccess_(result);
        updateLedgerLink_(result);
        // 로컬 대장 사본도 파일링크까지 기록한 최신 상태로 내보낸다.
        exportLedgerToAgent_(system, Number(result.year), result.quoteNumber);
        setQuoteOperationStatus_(result.details.operationId, 'completed', result.quoteNumber);
        Logger.log('[성능] 대장/링크 반영 ' + result.quoteNumber + ': ' + (Date.now() - ledgerStartedAt) + 'ms');
        Logger.log('대장 파일링크 갱신 완료(' + result.quoteNumber + '): ' + result.fileUrl);
      } catch (err) {
        setQuoteOperationStatus_(result.details && result.details.operationId, 'failed', result.quoteNumber, err.toString());
        Logger.log('[성능] 대장/링크 실패 ' + result.quoteNumber + ': ' + (Date.now() - ledgerStartedAt) + 'ms');
        // 파일은 이미 생성되었지만 대장 링크 갱신에 실패한 상태 — 결과를 지우면 영구히 유실되므로
        // resultsFailed 폴더에 원본을 보존하고, 매분 트리거가 같은 오류로 계속 재시도하지 않도록
        // results 폴더에서는 제거한다. 관리자가 수동으로 확인/재처리해야 한다.
        Logger.log('대장 파일링크 갱신 실패(' + result.quoteNumber + '): ' + err.toString());
        try {
          getAgentSubfolder_('resultsFailed').createFile(
            `링크실패_${sanitizeDriveName(result.quoteNumber || file.getName())}_${Date.now()}.json`,
            JSON.stringify({ result: result, error: err.toString() }),
            MimeType.PLAIN_TEXT
          );
        } catch (copyErr) {
          Logger.log('resultsFailed 보존 실패: ' + copyErr.toString());
        }
      }
    } else if (result && result.ok) {
      // 성공 보고인데 필수 값이 빠진 경우에는 조용히 삭제하지 않는다.
      // fileUrl이 비어 있으면 대장 링크를 만들 수 없으므로 원본 결과를 보존해야 한다.
      const missing = [];
      if (!result.department) missing.push('department');
      if (!result.quoteNumber) missing.push('quoteNumber');
      if (!result.fileUrl) missing.push('fileUrl');
      const message = '완료 보고 필수 값 누락: ' + missing.join(', ');
      setQuoteOperationStatus_(result.details && result.details.operationId, 'failed', result.quoteNumber, message);
      Logger.log('대장 파일링크 갱신 보류(' + (result.quoteNumber || file.getName()) + '): ' + message);
      try {
        getAgentSubfolder_('resultsFailed').createFile(
          `링크실패_${sanitizeDriveName(result.quoteNumber || file.getName())}_${Date.now()}.json`,
          JSON.stringify({ result: result, error: message }),
          MimeType.PLAIN_TEXT
        );
      } catch (copyErr) {
        Logger.log('resultsFailed 보존 실패: ' + copyErr.toString());
      }
    } else if (result && result.ok === false) {
      setQuoteOperationStatus_(result.operationId, 'failed', result.quoteNumber, result.error || '견적 파일 생성에 실패했습니다.');
      movePendingToFailed_(result.quoteNumber, result.error || '알 수 없는 오류');
    }
    permanentlyDeleteFile_(file.getId());
  }
}

function quoteOperationStatusKey_(operationId) {
  return 'QUOTE_OPERATION_' + String(operationId || '').trim();
}

function setQuoteOperationStatus_(operationId, status, quoteNumber, message) {
  const id = String(operationId || '').trim();
  if (!id) return;
  try {
    CacheService.getScriptCache().put(quoteOperationStatusKey_(id), JSON.stringify({
      success: true,
      status: status,
      quoteNumber: String(quoteNumber || ''),
      message: String(message || '')
    }), 21600);
  } catch (err) {
    Logger.log('견적 작업 상태 캐시 기록 실패(' + id + '): ' + err.toString());
  }
}

// 저장 완료 대기 화면용 경량 API. 대장 전체 조회 대신 큐 결과와 operationId만 확인한다.
function getQuoteOperationStatusFromReact(payload) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, status: 'failed', message: '견적 접근 권한을 확인할 수 없습니다.' };
    }

    const operationId = String(payload && payload.operationId || '').trim();
    if (!operationId) {
      return { success: false, status: 'failed', message: '견적 작업 ID가 없습니다.' };
    }

    const requestedDepartment = String(payload && payload.department || '').trim();
    let department = access.author.department;
    if (access.isAdmin && CONFIG.DEPARTMENT_ROOTS[requestedDepartment]) {
      department = requestedDepartment;
    } else if (requestedDepartment && requestedDepartment !== department) {
      return { success: false, status: 'failed', message: '다른 부서의 견적 작업을 조회할 권한이 없습니다.' };
    }

    const cache = CacheService.getScriptCache();
    const statusKey = quoteOperationStatusKey_(operationId);
    let cached = cache.get(statusKey);
    if (cached) {
      try {
        const status = JSON.parse(cached);
        if (status.status === 'completed' || status.status === 'failed') return status;
      } catch (err) { /* 손상된 상태 캐시는 다시 조회한다. */ }
    }

    // 동시에 들어오는 폴링은 2초에 한 번만 Drive results 큐를 확인한다.
    const pollKey = 'QUOTE_OPERATION_RESULT_POLL_' + department;
    if (!cache.get(pollKey)) {
      cache.put(pollKey, '1', 2);
      processAgentResults(undefined, department);
    }

    cached = cache.get(statusKey);
    if (cached) {
      try {
        const status = JSON.parse(cached);
        if (status.status === 'completed' || status.status === 'failed') return status;
      } catch (err) { /* 손상된 캐시는 pending 응답으로 처리한다. */ }
    }

    // CacheService가 상태를 퇴거한 경우 스냅샷에서 operationId를 한 번 복구 조회한다.
    if (!cached) {
      const currentYear = new Date().getFullYear();
      const parsedYear = Number(payload && payload.year);
      const year = Number.isInteger(parsedYear) && parsedYear >= 2000 && parsedYear <= currentYear
        ? parsedYear
        : currentYear;
      const system = getExistingYearSystem_(year, resolveRootFolderId({ authorDepartment: department }));
      const snapshot = system ? getQuoteSnapshotSheet_(system.ledger, false) : null;
      if (snapshot && snapshot.getLastRow() >= 2) {
        const match = snapshot.getRange(2, 5, snapshot.getLastRow() - 1, 1)
          .createTextFinder(operationId)
          .matchCase(true)
          .findNext();
        if (match) {
          try {
            const stored = JSON.parse(String(match.getDisplayValue() || '{}'));
            if (String(stored.details && stored.details.operationId || '') === operationId) {
              const completed = {
                success: true,
                status: 'completed',
                quoteNumber: String(stored.details.quoteNumber || '')
              };
              cache.put(statusKey, JSON.stringify(completed), 21600);
              return completed;
            }
          } catch (err) { /* 손상된 스냅샷은 완료로 오인하지 않는다. */ }
        }
      }
    }

    const pending = { success: true, status: 'pending', quoteNumber: String(payload && payload.quoteNumber || '') };
    if (!cached) cache.put(statusKey, JSON.stringify(pending), 21600);
    return pending;
  } catch (err) {
    return { success: false, status: 'failed', message: '견적 처리 상태 조회 실패: ' + err.toString() };
  }
}

// 대장 열 구성 오류로 resultsFailed에 보존된 견적 결과를, 열 수정 후 안전하게 재반영한다.
// Apps Script 편집기에서 recoverFailedLedgerResultsAfterLayoutFix를 실행한다.
// 동일 견적번호의 중복 실패본은 가장 최근 결과 하나만 처리하며, 원본 실패 JSON은 감사용으로 보존한다.
function recoverFailedLedgerResultsAfterLayoutFix() {
  const failedFolder = getAgentSubfolder_('resultsFailed');
  const iterator = failedFolder.getFiles();
  const latestByQuote = {};

  while (iterator.hasNext()) {
    const file = iterator.next();
    if (file.getName().indexOf('링크실패_') !== 0) continue;
    let wrapper;
    try {
      wrapper = JSON.parse(file.getBlob().getDataAsString('utf8'));
    } catch (err) {
      Logger.log('[대장 복구 건너뜀] JSON 파싱 실패: ' + file.getName());
      continue;
    }

    const result = wrapper && wrapper.result;
    const errorText = String(wrapper && wrapper.error || '');
    if (!result || result.ok !== true || errorText.indexOf('열 구성이 예상과 다릅니다') === -1) continue;
    if (!result.department || !result.quoteNumber || !result.fileUrl) continue;

    const key = [result.department, result.year, result.quoteNumber].join('|');
    const modifiedAt = file.getLastUpdated().getTime();
    if (!latestByQuote[key] || modifiedAt > latestByQuote[key].modifiedAt) {
      latestByQuote[key] = { result: result, modifiedAt: modifiedAt, fileName: file.getName() };
    }
  }

  const candidates = Object.keys(latestByQuote).map(function(key) { return latestByQuote[key]; });
  if (candidates.length === 0) {
    Logger.log('열 구성 오류로 재처리할 견적 결과가 없습니다.');
    return { success: true, recovered: 0, failed: 0 };
  }

  // 모든 대상 대장의 헤더를 먼저 검증해, 일부만 반영된 뒤 다음 건에서 헤더 오류가 나지 않도록 한다.
  candidates.forEach(function(candidate) {
    const result = candidate.result;
    const rootFolderId = resolveRootFolderId({ authorDepartment: result.department });
    const system = getExistingYearSystem_(Number(result.year), rootFolderId);
    if (!system) throw new Error('대장을 찾을 수 없습니다: ' + result.department + '/' + result.year);
    assertLedgerLayout_(system.ledgerSheet);
  });

  let recovered = 0;
  let failed = 0;
  candidates.forEach(function(candidate) {
    const result = candidate.result;
    try {
      const system = commitLedgerAfterAgentSuccess_(result);
      updateLedgerLink_(result);
      exportLedgerToAgent_(system, Number(result.year), result.quoteNumber);
      setQuoteOperationStatus_(result.details && result.details.operationId, 'completed', result.quoteNumber);
      recovered++;
      Logger.log('[대장 복구 완료] ' + result.department + ' / ' + result.quoteNumber + ' (원본: ' + candidate.fileName + ')');
    } catch (err) {
      failed++;
      Logger.log('[대장 복구 실패] ' + result.department + ' / ' + result.quoteNumber + ': ' + err.toString());
    }
  });

  Logger.log('실패 결과 복구 완료: 성공 ' + recovered + '건, 실패 ' + failed + '건');
  return { success: failed === 0, recovered: recovered, failed: failed };
}

function updateLedgerLink_(result) {
  const rootFolderId = resolveRootFolderId({ authorDepartment: result.department });
  const year = Number(result.year) || new Date().getFullYear();
  const system = getExistingYearSystem_(year, rootFolderId);
  if (!system) {
    throw new Error('대장을 찾을 수 없습니다(부서: ' + result.department + ', 연도: ' + year + ')');
  }
  const ledger = system.ledgerSheet;
  const ledgerQuoteNumber = String(result.ledgerQuoteNumber || baseQuoteNumber_(result.quoteNumber)).trim();
  const row = findLedgerRow_(ledger, ledgerQuoteNumber);
  if (row < 2) {
    throw new Error('대장에서 견적번호를 찾을 수 없습니다: ' + ledgerQuoteNumber);
  }

  const fileUrl = String(result.fileUrl);
  const path = fileUrl.split('?')[0].split('/');
  const fileName = decodeURIComponent(path[path.length - 1] || '열기');
  const revisionNumber = Number(result.revisionNumber) || 0;
  const linkLabel = revisionNumber > 0
    ? `Rev${String(revisionNumber).padStart(2, '0')}: ${fileName}`
    : `원본: ${fileName}`;
  const cell = ledger.getRange(row, LEDGER_COL.FILE_LINK);
  const links = [];
  const existing = cell.getRichTextValue();
  if (existing) {
    existing.getRuns().forEach(function(run) {
      const url = run.getLinkUrl();
      if (!url) return; // 구분자(' | ')처럼 링크가 없는 런은 누적하지 않는다.
      const text = run.getText();
      if (text) links.push({ text: text, url: url });
    });
  }
  if (links.length === 0) {
    const currentText = String(cell.getDisplayValue() || '').trim();
    if (currentText) links.push({ text: currentText, url: null });
  }
  // 덮어쓰기 지원: 동일 파일명이나 라벨이 이미 존재하면 새 URL로 교체하고, 없으면 추가
  const existingIdx = links.findIndex(function(link) {
    return link.url === fileUrl || link.text === linkLabel || link.text.indexOf(fileName) !== -1;
  });
  if (existingIdx >= 0) {
    links[existingIdx] = { text: linkLabel, url: fileUrl };
  } else {
    links.push({ text: linkLabel, url: fileUrl });
  }

  const separator = ' | ';
  const combinedText = links.map(function(link) { return link.text; }).join(separator);
  const builder = SpreadsheetApp.newRichTextValue().setText(combinedText);
  let offset = 0;
  links.forEach(function(link) {
    if (link.url) builder.setLinkUrl(offset, offset + link.text.length, link.url);
    offset += link.text.length + separator.length;
  });
  cell.setRichTextValue(builder.build());
}

function movePendingToFailed_(quoteNumber, error) {
  try {
    const pending = getAgentSubfolder_('pending');
    const failed = getAgentSubfolder_('failed');
    const iter = pending.getFilesByName(`${sanitizeDriveName(quoteNumber)}.json`);
    if (iter.hasNext()) {
      const file = iter.next();
      failed.createFile(
        `실패_${sanitizeDriveName(quoteNumber)}_${Date.now()}.json`,
        JSON.stringify({ quoteNumber: quoteNumber, error: error }),
        MimeType.PLAIN_TEXT
      );
      file.setTrashed(true);
    }
  } catch (err) {
    Logger.log('실패 폴더 이동 오류(' + quoteNumber + '): ' + err.toString());
  }
}

function openAuthorDatabaseSpreadsheet_() {
  return SpreadsheetApp.openById(CONFIG.AUTHOR_DB_SHEET_ID);
}

function normalizeAuthorDepartment_(value) {
  const raw = String(value || '').trim();
  const key = raw.replace(/[\s\u00A0]/g, '');
  const aliases = {
    '기술영업': '기술영업',
    '기술영업실': '기술영업',
    '영업': '영업',
    '자동화영업실': '영업',
    '자동화영업': '영업',
    '프로젝트': '프로젝트',
    '프로젝트사업실': '프로젝트',
    '프로젝트사업': '프로젝트'
  };
  return aliases[key] || raw;
}

function normalizeAccountEmail_(value) {
  const raw = String(value || '').replace(/[\u200B-\u200D\uFEFF\u00A0]/g, ' ').trim();
  const wrapped = raw.match(/<([^<>]+)>/);
  return String(wrapped ? wrapped[1] : raw).trim().toLowerCase();
}

// 작성자 DB 시트(시트1)에서 행 목록을 읽는 공용 헬퍼.
// 기대하는 열 구성: 작성자(또는 이름/성명) | 연락처 | 이메일 | 부서
// 1행 헤더의 키워드로 열 위치를 찾고, 헤더가 없으면 A~D열 고정 순서로 해석한다.
function readAuthorRows_() {
  const spreadsheet = openAuthorDatabaseSpreadsheet_();
  const sheet = spreadsheet.getSheetByName(CONFIG.AUTHOR_DB_SHEET_NAME) || spreadsheet.getSheets()[0];
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  if (lastRow < 1 || lastColumn < 1) return [];

  const values = sheet.getRange(1, 1, lastRow, lastColumn).getValues();
  const headers = values[0].map(function(v) { return String(v || '').trim(); });

  function findColumn(keywords) {
    for (let c = 0; c < headers.length; c++) {
      const header = String(headers[c] || '').toLowerCase();
      for (const keyword of keywords) {
        if (header.indexOf(String(keyword).toLowerCase()) !== -1) return c;
      }
    }
    return -1;
  }

  let nameCol = findColumn(['작성자', '이름', '성명', 'name', 'author']);
  let phoneCol = findColumn(['연락처', '전화', '휴대폰', 'phone', 'mobile', 'tel']);
  let emailCol = findColumn(['이메일', '메일', 'email', 'e-mail']);
  let deptCol = findColumn(['부서', '소속', '팀', 'department', 'dept', 'team']);

  const headerMatched = nameCol !== -1;
  if (!headerMatched) {
    nameCol = 0;
    phoneCol = 1;
    emailCol = 2;
    deptCol = 3;
  }

  const rows = [];
  for (let r = headerMatched ? 1 : 0; r < values.length; r++) {
    const name = String(values[r][nameCol] || '').trim();
    if (!name) continue;
    rows.push({
      name: name,
      phone: phoneCol >= 0 ? String(values[r][phoneCol] || '').trim() : '',
      email: emailCol >= 0 ? normalizeAccountEmail_(values[r][emailCol]) : '',
      department: deptCol >= 0 ? normalizeAuthorDepartment_(values[r][deptCol]) : ''
    });
  }
  return rows;
}

// 작성자 DB 시트(시트1)에서 작성자 목록을 읽는다.
function getAuthorsFromReact() {
  try {
    return { success: true, authors: readAuthorRows_() };
  } catch (err) {
    return { success: false, message: '작성자 목록을 불러오지 못했습니다: ' + err.toString(), authors: [] };
  }
}

// 접속 중인 Google 계정이 작성자 DB에 등록되어 있는지 확인한다.
// 견적 기능 접근 권한 확인 + 부서 조회 용도로 사용된다.
function getAuthorizedUserFromReact() {
  let userEmail = '';
  try {
    userEmail = normalizeAccountEmail_(Session.getActiveUser().getEmail());
    if (!userEmail) {
      return { success: true, authorized: false, email: '', message: 'Google 계정을 확인할 수 없습니다.' };
    }
    const rows = readAuthorRows_();
    const matchingRows = rows.filter(function(row) {
      return normalizeAccountEmail_(row.email) === userEmail;
    });
    if (matchingRows.length === 0) {
      return { success: true, authorized: false, email: userEmail };
    }
    const isAdmin = Boolean(CONFIG.ADMIN_EMAILS && CONFIG.ADMIN_EMAILS.some(function(adminEmail) {
      return normalizeAccountEmail_(adminEmail) === userEmail;
    }));
    const registeredDepartments = Array.from(new Set(matchingRows.map(function(row) {
      return normalizeAuthorDepartment_(row.department);
    })));
    if (!isAdmin && registeredDepartments.length > 1) {
      return {
        success: true,
        authorized: false,
        email: userEmail,
        message: '작성자 DB에 동일한 이메일이 여러 부서로 등록되어 있습니다. 한 부서로 정리한 뒤 다시 시도해 주세요.'
      };
    }
    const matched = matchingRows[0];
    const department = normalizeAuthorDepartment_(matched.department);
    if (!isAdmin && !CONFIG.DEPARTMENT_ROOTS[department]) {
      return {
        success: true,
        authorized: false,
        email: userEmail,
        message: '작성자 DB의 부서 값이 견적 시스템 부서 목록과 일치하지 않습니다. 관리자에게 부서 등록을 확인해 주세요.'
      };
    }
    const author = Object.assign({}, matched, {
      department: CONFIG.DEPARTMENT_ROOTS[department] ? department : CONFIG.DEFAULT_DEPARTMENT
    });
    return {
      success: true,
      authorized: true,
      author: author,
      isAdmin: isAdmin,
      availableDepartments: isAdmin ? Object.keys(CONFIG.DEPARTMENT_ROOTS) : [department]
    };
  } catch (err) {
    Logger.log('권한 확인 중 오류(' + (userEmail || '계정 이메일 확인 불가') + '): ' + err.toString());
    return {
      success: false,
      authorized: false,
      email: userEmail,
      message: '권한 확인 중 오류가 발생했습니다: ' + err.toString()
    };
  }
}

function resolveRootFolderId(details) {
  const department = String((details && details.authorDepartment) || '').trim();
  return CONFIG.DEPARTMENT_ROOTS[department] || CONFIG.DEPARTMENT_ROOTS[CONFIG.DEFAULT_DEPARTMENT];
}

function getYearSystem(year, rootFolderId) {
  const targetRootId = rootFolderId || CONFIG.DEPARTMENT_ROOTS[CONFIG.DEFAULT_DEPARTMENT];
  const rootFolder = DriveApp.getFolderById(targetRootId);
  const yearStr = year.toString();

  let yearFolder;
  const folderIter = rootFolder.getFoldersByName(yearStr);
  if (folderIter.hasNext()) {
    yearFolder = folderIter.next();
  } else {
    yearFolder = rootFolder.createFolder(yearStr);
  }

  const ledgerName = `${yearStr}_견적관리대장`;
  let ledgerSS;
  const fileIter = yearFolder.getFilesByName(ledgerName);
  if (fileIter.hasNext()) {
    ledgerSS = SpreadsheetApp.open(fileIter.next());
  } else {
    // 새 연도가 시작되면 "견적관리대장 샘플" 템플릿을 복사해 해당 연도 폴더에 만든다.
    const ledgerTemplate = DriveApp.getFileById(CONFIG.LEDGER_TEMPLATE_ID);
    const newLedgerFile = ledgerTemplate.makeCopy(ledgerName, yearFolder);
    ledgerSS = SpreadsheetApp.open(newLedgerFile);
  }

  const ledgerSheet = ledgerSS.getSheetByName(CONFIG.LEDGER_SHEET_NAME) || ledgerSS.getSheets()[0];
  return { folder: yearFolder, ledger: ledgerSS, ledgerSheet };
}

// 목록 조회에서는 존재하지 않는 연도의 폴더·대장을 새로 만들지 않는다.
function getExistingYearSystem_(year, rootFolderId) {
  const targetRootId = rootFolderId || CONFIG.DEPARTMENT_ROOTS[CONFIG.DEFAULT_DEPARTMENT];
  const rootFolder = DriveApp.getFolderById(targetRootId);
  const yearStr = String(year);
  const folderIter = rootFolder.getFoldersByName(yearStr);
  if (!folderIter.hasNext()) return null;

  const ledgerName = `${yearStr}_견적관리대장`;
  const fileIter = yearFolder.getFilesByName(ledgerName);
  if (!fileIter.hasNext()) return null;
  const ledgerSS = SpreadsheetApp.open(fileIter.next());
  const ledgerSheet = ledgerSS.getSheetByName(CONFIG.LEDGER_SHEET_NAME) || ledgerSS.getSheets()[0];
  return { folder: yearFolder, ledger: ledgerSS, ledgerSheet };
}

function getAvailableLedgerYears_(rootFolderId, currentYear) {
  const targetRootId = rootFolderId || CONFIG.DEPARTMENT_ROOTS[CONFIG.DEFAULT_DEPARTMENT];
  // 매 조회마다 드라이브의 모든 연도 폴더와 파일을 전수 순회하지 않도록 CacheService로 캐싱(10분)
  const cacheKey = `YEARS_${targetRootId}_${currentYear}`;
  const cache = CacheService.getScriptCache();
  const cached = cache.get(cacheKey);
  if (cached) {
    try {
      return JSON.parse(cached);
    } catch (e) { /* 캐시 파싱 실패 시 재조회 */ }
  }

  const rootFolder = DriveApp.getFolderById(targetRootId);
  const folders = rootFolder.getFolders();
  const years = [];
  while (folders.hasNext()) {
    const folder = folders.next();
    const yearName = String(folder.getName()).trim();
    if (!/^\d{4}$/.test(yearName)) continue;
    const year = Number(yearName);
    if (year > currentYear) continue;
    const ledgerFiles = folder.getFilesByName(`${yearName}_견적관리대장`);
    if (ledgerFiles.hasNext()) years.push(year);
  }
  const result = years.sort((left, right) => right - left);
  try {
    cache.put(cacheKey, JSON.stringify(result), 600); // 10분간 캐시
  } catch (e) { /* 캐시 저장 실패해도 무해 */ }
  return result;
}

function baseQuoteNumber_(quoteNumber) {
  return String(quoteNumber || '').trim().replace(/_Rev\d+$/i, '');
}

function findLedgerRow_(sheet, quoteNumber) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;
  const numbers = sheet.getRange(2, LEDGER_COL.QUOTE_NUMBER, lastRow - 1, 1).getDisplayValues();
  const target = String(quoteNumber || '').trim();
  for (let i = 0; i < numbers.length; i++) {
    if (String(numbers[i][0] || '').trim() === target) return i + 2;
  }
  return -1;
}

function getNextRevisionNumber_(spreadsheet, ledgerSheet, baseQuoteNumber) {
  let maxRevision = 0;
  const snapshot = getQuoteSnapshotSheet_(spreadsheet, false);
  if (snapshot && snapshot.getLastRow() >= 2) {
    const rows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 2).getDisplayValues();
    rows.forEach(function(row) {
      if (String(row[1] || '').trim() !== baseQuoteNumber) return;
      const match = String(row[0] || '').match(/_Rev(\d+)$/i);
      if (match) maxRevision = Math.max(maxRevision, parseInt(match[1], 10));
    });
  }

  const ledgerRow = findLedgerRow_(ledgerSheet, baseQuoteNumber);
  if (ledgerRow >= 2) {
    const linkText = String(ledgerSheet.getRange(ledgerRow, LEDGER_COL.FILE_LINK).getDisplayValue() || '');
    const matches = linkText.match(/_Rev(\d+)/gi) || [];
    matches.forEach(function(value) {
      const match = value.match(/(\d+)/);
      if (match) maxRevision = Math.max(maxRevision, parseInt(match[1], 10));
    });
  }
  return maxRevision + 1;
}

function saveQuoteSnapshot_(spreadsheet, details, items, totalPrice) {
  const snapshot = getQuoteSnapshotSheet_(spreadsheet, true);
  const outputQuoteNumber = String(details.quoteNumber || '').trim();
  const baseQuoteNumber = String(details.baseQuoteNumber || outputQuoteNumber).trim();
  if (!outputQuoteNumber) return;

  const snapshotData = JSON.stringify({ details: details, items: items || [], totalPrice: totalPrice });
  const now = new Date();
  const authorEmail = String(details.authorEmail || '').trim();

  // 기존 스냅샷이 이미 존재하면(덮어쓰기 또는 완료 보고 재처리 시) 해당 행을 최신 데이터로 업데이트!
  if (snapshot.getLastRow() >= 2) {
    const existingNumbers = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 1).getDisplayValues();
    for (let i = 0; i < existingNumbers.length; i++) {
      if (String(existingNumbers[i][0] || '').trim() === outputQuoteNumber) {
        snapshot.getRange(i + 2, 3, 1, 3).setValues([[now, authorEmail, snapshotData]]);
        return;
      }
    }
  }
  snapshot.appendRow([
    outputQuoteNumber,
    baseQuoteNumber,
    now,
    authorEmail,
    snapshotData
  ]);
}

function getQuoteFolderClientName_(spreadsheet, baseQuoteNumber) {
  const snapshot = getQuoteSnapshotSheet_(spreadsheet, false);
  if (!snapshot || snapshot.getLastRow() < 2) return '';
  const rows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues();
  for (let i = rows.length - 1; i >= 0; i--) {
    if (String(rows[i][1] || '').trim() !== baseQuoteNumber) continue;
    try {
      const stored = JSON.parse(String(rows[i][4] || '{}'));
      return String(stored.details?.folderClientName || stored.details?.clientName || '').trim();
    } catch (err) {
      return '';
    }
  }
  return '';
}

function getQuoteForEditFromReact(requestedYear, quoteNumber, requestedDepartment) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '견적 수정 권한을 확인할 수 없습니다.' };
    }
    const currentYear = new Date().getFullYear();
    const parsedYear = Number(requestedYear);
    const year = Number.isInteger(parsedYear) && parsedYear >= 2000 && parsedYear <= currentYear
      ? parsedYear
      : currentYear;

    const rawQuoteNum = String(quoteNumber || '').trim();
    if (!rawQuoteNum) {
      return { success: false, message: '수정할 견적번호가 비어 있습니다.' };
    }

    const baseQuoteNumber = baseQuoteNumber_(rawQuoteNum);
    const prefixDept = rawQuoteNum.split(/\s+/)[0];
    let department = access.author.department;
    if (CONFIG.DEPARTMENT_ROOTS[prefixDept]) {
      department = prefixDept;
    } else if (access.isAdmin && requestedDepartment && CONFIG.DEPARTMENT_ROOTS[String(requestedDepartment).trim()]) {
      department = String(requestedDepartment).trim();
    }

    const rootFolderId = resolveRootFolderId({ authorDepartment: department });
    const system = getExistingYearSystem_(year, rootFolderId);
    if (!system) {
      return { success: false, message: `수정할 ${year}년 ${department} 대장을 찾을 수 없습니다.` };
    }

    const sheet = system.ledgerSheet;
    const ledgerRowIdx = findLedgerRow_(sheet, baseQuoteNumber);

    // 정규화 키 함수 (공백, 하이픈, 언더바 제거)
    function normKey(str) {
      return String(str || '').replace(/[\s\-_]/g, '').toLowerCase();
    }
    const baseNorm = normKey(baseQuoteNumber);

    const existingVersions = [];
    const seenVerNumbers = {};

    // 1. 스냅샷 시트(견적원본데이터)에서 정밀 품목 데이터 탐색
    const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
    if (snapshot && snapshot.getLastRow() >= 2) {
      const rows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues();

      // 기존 버전 목록 수집
      for (let j = 0; j < rows.length; j++) {
        const q0 = String(rows[j][0] || '').trim();
        const q1 = String(rows[j][1] || '').trim();
        if (normKey(q1) !== baseNorm && normKey(q0).indexOf(baseNorm) === -1) continue;
        const verNum = q0 || q1;
        if (!verNum || seenVerNumbers[verNum]) continue;
        seenVerNumbers[verNum] = true;
        const isRev = /_Rev(\d+)$/i.test(verNum);
        const revTag = isRev ? verNum.match(/_Rev(\d+)$/i)[0].replace(/^_/, '') : '원본';
        const dateStr = rows[j][2] ? Utilities.formatDate(new Date(rows[j][2]), 'GMT+9', 'yyyy-MM-dd HH:mm') : '';
        existingVersions.push({
          quoteNumber: verNum,
          label: `${revTag} (${verNum})`,
          date: dateStr
        });
      }

      // 최신 스냅샷 데이터 탐색 (역순)
      for (let i = rows.length - 1; i >= 0; i--) {
        const q0 = String(rows[i][0] || '').trim();
        const q1 = String(rows[i][1] || '').trim();
        if (normKey(q1) !== baseNorm && normKey(q0).indexOf(baseNorm) === -1) continue;

        try {
          const stored = JSON.parse(String(rows[i][4] || '{}'));
          if (stored && (stored.details || stored.items)) {
            if (!seenVerNumbers[baseQuoteNumber]) {
              existingVersions.unshift({
                quoteNumber: baseQuoteNumber,
                label: `원본 (${baseQuoteNumber})`,
                date: ''
              });
            }
            return {
              success: true,
              quote: {
                quoteNumber: q0 || stored.details?.quoteNumber || baseQuoteNumber,
                baseQuoteNumber: baseQuoteNumber,
                year: year,
                department: department,
                details: stored.details || {},
                items: Array.isArray(stored.items) ? stored.items : [],
                existingVersions: existingVersions
              }
            };
          }
        } catch (parseErr) {
          Logger.log('스냅샷 파싱 실패: ' + parseErr.toString());
        }
      }
    }

    // 2. [Fallback 복원] 스냅샷이 없는 경우(과거 수동 작성 440건 등):
    // 대장 시트의 해당 행 데이터로부터 기본 견적 정보 및 품목을 복원하여 열어준다!
    if (ledgerRowIdx >= 2) {
      const rowValues = sheet.getRange(ledgerRowIdx, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];
      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];

      function colVal(labels, fallbackCol) {
        const idx = headers.findIndex(function(h) {
          return labels.some(function(l) { return String(h || '').indexOf(l) !== -1; });
        });
        return idx >= 0 ? String(rowValues[idx] || '').trim() : (fallbackCol ? String(rowValues[fallbackCol - 1] || '').trim() : '');
      }

      const clientName = colVal(['업체명', '회사명'], LEDGER_COL.CLIENT_NAME);
      const contactPerson = colVal(['고객명', '담당자'], LEDGER_COL.CLIENT_CONTACT);
      const clientPhone = colVal(['연락처', '전화'], LEDGER_COL.CLIENT_PHONE);
      const clientEmail = colVal(['이메일', '메일'], LEDGER_COL.CLIENT_EMAIL);
      const productName = colVal(['제품명'], LEDGER_COL.PRODUCT_NAME);
      const categories = colVal(['제품 항목', '제품군'], LEDGER_COL.CATEGORIES);
      const rawPrice = colVal(['견적 금액', '금액'], LEDGER_COL.TOTAL_PRICE);
      const totalPrice = Number(rawPrice.replace(/[^\d.-]/g, '')) || 0;
      const notes = colVal(['비고'], LEDGER_COL.NOTES);
      const quoteDate = `${rowValues[LEDGER_COL.YEAR - 1]}.${rowValues[LEDGER_COL.MONTH - 1]}.${rowValues[LEDGER_COL.DAY - 1]}`;

      existingVersions.push({
        quoteNumber: baseQuoteNumber,
        label: `원본 (${baseQuoteNumber})`,
        date: quoteDate
      });

      return {
        success: true,
        quote: {
          quoteNumber: baseQuoteNumber,
          baseQuoteNumber: baseQuoteNumber,
          year: year,
          department: department,
          details: {
            quoteNumber: baseQuoteNumber,
            quoteDate: quoteDate,
            clientName: clientName,
            clientContactPerson: contactPerson,
            clientPhone: clientPhone,
            clientEmail: clientEmail,
            notes: notes,
            authorName: colVal(['작성자'], LEDGER_COL.AUTHOR) || access.author.name || '',
            authorPhone: access.author.phone || '',
            authorEmail: access.author.email || '',
            authorDepartment: department
          },
          items: [{
            type: categories || '기타',
            name: productName || '견적 제품',
            spec: '',
            quantity: 1,
            unitPrice: totalPrice,
            multiplier: 1,
            totalPrice: totalPrice
          }],
          existingVersions: existingVersions
        }
      };
    }

    return {
      success: false,
      message: `수정할 견적 원본을 찾을 수 없습니다(견적번호: ${baseQuoteNumber}, 연도: ${year}, 부서: ${department}).`
    };
  } catch (err) {
    return { success: false, message: '견적 원본 조회 실패: ' + err.toString() };
  }
}

function updateQuoteOrderFromReact(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (err) {
    return { success: false, message: '다른 사용자가 대장을 갱신 중입니다. 잠시 후 다시 시도해 주세요.' };
  }
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '발주 상태를 변경할 권한이 없습니다.' };
    }
    const currentYear = new Date().getFullYear();
    const parsedYear = Number(payload && payload.year);
    const year = Number.isInteger(parsedYear) && parsedYear >= 2000 && parsedYear <= currentYear
      ? parsedYear
      : currentYear;

    const baseQuoteNumber = baseQuoteNumber_(payload && payload.quoteNumber);
    const prefixDept = String(baseQuoteNumber || '').trim().split(/\s+/)[0];
    let department = access.author.department;
    if (CONFIG.DEPARTMENT_ROOTS[prefixDept]) {
      department = prefixDept;
    } else if (access.isAdmin && payload && payload.department && CONFIG.DEPARTMENT_ROOTS[String(payload.department).trim()]) {
      department = String(payload.department).trim();
    }

    const rootFolderId = resolveRootFolderId({ authorDepartment: department });
    const system = getExistingYearSystem_(year, rootFolderId);
    if (!system || !baseQuoteNumber) {
      return { success: false, message: '발주 상태를 기록할 견적을 찾을 수 없습니다.' };
    }
    const row = findLedgerRow_(system.ledgerSheet, baseQuoteNumber);
    if (row < 2) return { success: false, message: '발주 상태를 기록할 견적을 찾을 수 없습니다.' };
    // LEDGER_COL.ORDER(E열)를 우선 사용하고, 헤더가 실제로 '발주'를 포함하는지 검증한다.
    // 대장 구조가 예상과 다를 때 엉뚱한 열을 덮어쓰지 않도록 검증 실패 시 전체 헤더에서 재탐색한다.
    const headers = system.ledgerSheet.getRange(1, 1, 1, system.ledgerSheet.getLastColumn()).getDisplayValues()[0];
    let orderColumn = String(headers[LEDGER_COL.ORDER - 1] || '').indexOf('발주') !== -1 ? LEDGER_COL.ORDER - 1 : -1;
    if (orderColumn < 0) {
      orderColumn = headers.findIndex(function(header) {
        return String(header || '').indexOf('발주') !== -1;
      });
    }
    if (orderColumn < 0) {
      return { success: false, message: '견적관리대장에 발주 컬럼이 없습니다.' };
    }
    system.ledgerSheet.getRange(row, orderColumn + 1).setValue(payload && payload.ordered === true ? '발주' : '');
    enqueueActivityLog_(
      access.author.email || access.author.name || department,
      '발주',
      `견적번호: ${baseQuoteNumber} | 상태: ${(payload && payload.ordered === true) ? '발주 완료(체크)' : '발주 취소(체크 해제)'} | 부서: ${department}`
    );
    return { success: true, message: '발주 여부가 기록되었습니다.' };
  } catch (err) {
    return { success: false, message: '발주 상태 기록 실패: ' + err.toString() };
  } finally {
    lock.releaseLock();
  }
}

    const headers = sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0];
    let siteCol = headers.findIndex(function(h) {
      return /적용\s*현장|현장명|^현장$/.test(String(h || '').trim());
    });

    if (siteCol < 0) {
      const newColIndex = lastCol + 1;
      sheet.getRange(1, newColIndex).setValue('적용현장');
      siteCol = newColIndex - 1;
      SpreadsheetApp.flush();
    }

    const siteName = String((payload && payload.siteName) || '').trim();
    sheet.getRange(row, siteCol + 1).setValue(siteName);
    SpreadsheetApp.flush();

    enqueueActivityLog_(
      access.author.email || access.author.name || department,
      '적용현장',
      `견적번호: ${baseQuoteNumber} | 적용현장: ${siteName || '(삭제)'} | 부서: ${department}`
    );
    return { success: true, message: '적용 현장이 저장되었습니다.', siteName: siteName };
  } catch (err) {
    return { success: false, message: '적용 현장 저장 실패: ' + err.toString() };
  } finally {
    lock.releaseLock();
  }
}

// 발주등록 요청 메일 초안 생성 (현재 로그인한 담당자의 Gmail 임시보관함에 저장)
// payload: {
//   year, department, quoteNumber,
//   clientName, productName, contactName, contactPhone,
//   deliveryAddress,   // 납품 주소 (필수)
//   files: [ { name, mimeType, base64 } ]  // 1개 이상 필수 (발주서·사업자등록증 등)
// }
// 발주등록 요청 메일 초안 생성 (현재 로그인한 담당자의 Gmail 임시보관함에 저장)
// payload.files는 로컬 에이전트 order-email 페이지가 읽어 postMessage로 전달한 base64 첨부 목록이다.
function createOrderDraftFromReact(payload) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '발주등록 요청 메일 작성 권한이 없습니다.' };
    }

    const clientName = String((payload && payload.clientName) || '').trim();
    const productName = String((payload && payload.productName) || '').trim();
    const contactName = String((payload && payload.contactName) || '').trim();
    const contactPhone = String((payload && payload.contactPhone) || '').trim();
    const deliveryAddress = String((payload && payload.deliveryAddress) || '').trim();
    const files = Array.isArray(payload && payload.files)
      ? payload.files
      : [];

    if (!deliveryAddress) {
      return { success: false, message: '납품 주소를 입력해야 발주 메일을 작성할 수 있습니다.' };
    }
    if (files.length === 0) {
      return { success: false, message: '첨부할 파일(발주서, 사업자등록증 등)을 1개 이상 선택해 주세요.' };
    }
    if (files.length > 10) {
      return { success: false, message: '첨부 파일은 최대 10개까지 선택할 수 있습니다.' };
    }

    const quoteNumber = String((payload && payload.quoteNumber) || '').trim();
    const attachments = [];
    files.forEach(function(file) {
      const fileName = String(file && file.name || '첨부파일');
      const mimeType = String(file && file.mimeType || 'application/octet-stream');
      const base64 = String(file && file.base64 || '');
      if (!base64) throw new Error('첨부 파일 데이터가 비어 있습니다: ' + fileName);
      if (/[^A-Za-z0-9+/=]/.test(base64)) throw new Error('첨부 파일의 base64 데이터 형식이 올바르지 않습니다: ' + fileName);
      try {
        // Apps Script는 parseUint8Array를 제공하지 않으므로 base64Decode로 Blob 바이트를 만든다.
        const bytes = Utilities.base64Decode(base64);
        attachments.push(Utilities.newBlob(bytes, mimeType).setName(fileName));
      } catch (err) {
        throw new Error('첨부 파일 변환 실패(' + fileName + '): ' + err.toString());
      }
    });

    const subject = `[발주서] ${clientName}- ${productName} 발주서 공유`.trim();

    // &부서 표기: 기술영업→기술영업실, 영업→자동화영업실, 프로젝트→프로젝트사업실
    const deptName = String((payload && payload.department) || access.author.department || '').trim();
    const deptLabelMap = { '기술영업': '기술영업실', '영업': '자동화영업실', '프로젝트': '프로젝트사업실' };
    const deptLabel = deptLabelMap[deptName] || (deptName ? deptName + '실' : '기술영업실');

    const body = [
      '안녕하세요,',
      `${deptLabel} ${access.author.name || ''}입니다.`,
      '',
      `${clientName} ${productName} 발주서를 공유드립니다.`,
      '',
      '<납품정보>',
      '',
      `주소: ${deliveryAddress}`,
      `수령인: ${contactName}`,
      `연락처: ${contactPhone}`,
      '',
      '필요하신 추가 사항이 있으시면 연락 또는 회신 부탁드립니다.',
      '',
      '감사합니다.'
    ].join('\n');

    // 본인(로그인 사용자) 계정으로 HTML 모드 초안을 생성한다 (Gmail 서명 이미지 파일 보존).
    const htmlBody = textToHtmlBody_(body);
    GmailApp.createDraft(ORDER_EMAIL_TO, subject, body, {
      cc: ORDER_EMAIL_CC,
      attachments: attachments,
      name: 'CIMON',
      htmlBody: htmlBody
    });

    enqueueActivityLog_(
      access.author.email || access.author.name || '',
      '발주 메일',
      `견적번호: ${quoteNumber} | 업체명: ${clientName} | 주소: ${deliveryAddress} | 첨부 ${attachments.length}개 | 임시보관함 초안 생성`
    );

    return {
      success: true,
      message: '발주등록 요청 메일 초안이 임시보관함에 생성되었습니다. 최종 확인 후 발송해 주세요.'
    };
  } catch (err) {
    return { success: false, message: '발주등록 요청 메일 작성 실패: ' + err.toString() };
  }
}

// 발주등록 요청 메일 첨부용 파일 목록 조회 (구글 드라이브 '문서/<견적폴더명>' 미러 기준)
// payload: { year, department, quoteNumber, company }
function getQuoteFilesFromReact(payload) {
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '발주등록 요청 메일 사용 권한이 없습니다.' };
    }
    const quoteNumber = String((payload && payload.quoteNumber) || '').trim();
    const company = String((payload && payload.company) || '').trim();
    if (!quoteNumber) {
      return { success: true, files: [], message: '견적번호가 없습니다.' };
    }

    let docsRoot;
    try {
      docsRoot = getAgentSubfolder_('문서');
    } catch (driveErr) {
      return {
        success: false,
        message: '구글 드라이브 공유 폴더(견적에이전트) 접근 권한이 없습니다. 관리자에게 공유 폴더 접근 권한을 요청해 주세요. (' + driveErr.toString() + ')'
      };
    }
    const folderName = `${baseQuoteNumber_(quoteNumber)}_${company}`;
    let docsFolder = null;
    const folderIter = docsRoot.getFoldersByName(folderName);
    if (folderIter.hasNext()) {
      docsFolder = folderIter.next();
    } else {
      const prefix = baseQuoteNumber_(quoteNumber) + '_';
      const allFolders = docsRoot.getFolders();
      while (allFolders.hasNext()) {
        const candidate = allFolders.next();
        if (String(candidate.getName() || '').startsWith(prefix)) {
          docsFolder = candidate;
          break;
        }
      }
    }
    if (!docsFolder) {
      return { success: true, files: [], folder: folderName };
    }

    const files = [];
    const fileIter = docsFolder.getFiles();
    while (fileIter.hasNext()) {
      const file = fileIter.next();
      if (file.isTrashed()) continue;
      let size = 0;
      try { size = file.getBlob().getSize(); } catch (err) { size = 0; }
      files.push({ name: file.getName(), size: size });
    }
    files.sort(function(a, b) { return String(a.name).localeCompare(String(b.name), 'ko'); });
    return { success: true, folder: docsFolder.getName(), files: files };
  } catch (err) {
    return { success: false, message: '견적 폴더 파일 목록 조회 실패: ' + err.toString() };
  }
}

function getNextQuoteNumber(rootFolderId, department) {
  try {
    const today = new Date();
    const system = getYearSystem(today.getFullYear(), rootFolderId);
    const sheet = system.ledgerSheet;
    const departmentName = String(department || CONFIG.DEFAULT_DEPARTMENT).trim() || CONFIG.DEFAULT_DEPARTMENT;
    const yearPrefix = `${departmentName} ${Utilities.formatDate(today, Session.getScriptTimeZone(), 'yy')}`;
    const lastRow = sheet.getLastRow();

    // 영구삭제로 생긴 빈 번호는 가장 작은 빈 번호로 재사용한다.
    // processQuote의 LockService가 동시 저장을 직렬화하므로 같은 번호 중복은 발생하지 않는다.
    let seq = 1;
    if (lastRow > 1) {
      const nos = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
      const usedNumbers = {};
      nos.forEach(function(row) {
        const value = Number(row[0]);
        if (Number.isInteger(value) && value > 0) usedNumbers[value] = true;
      });
      while (usedNumbers[seq]) seq += 1;
    }

    const seqStr = String(seq).padStart(3, '0');
    return `${yearPrefix}${Utilities.formatDate(today, Session.getScriptTimeZone(), 'MM')}-${seqStr}`;
  } catch (err) {
    throw new Error('견적번호 생성 실패: ' + err.toString());
  }
}

// 견적 삭제 — mode: 'permanent'(영구삭제, 행+스냅샷+로컬 파일 삭제, 빈 번호 재사용)
//                'strikethrough'(라인삭제, 대장에 검은 취소선, 견적번호 유지)
// payload: { year, department, quoteNumber, mode }
function deleteQuoteFromReact(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (err) {
    return { success: false, message: '다른 사용자가 삭제 또는 저장 중입니다. 잠시 후 다시 시도해 주세요.' };
  }
  try {
    const access = getAuthorizedUserFromReact();
    if (!access.success || !access.authorized || !access.author) {
      return { success: false, message: '견적 삭제 권한이 없습니다.' };
    }
    const currentYear = new Date().getFullYear();
    const parsedYear = Number(payload && payload.year);
    const year = Number.isInteger(parsedYear) && parsedYear >= 2000 && parsedYear <= currentYear
      ? parsedYear : currentYear;
    const baseQuoteNumber = baseQuoteNumber_(payload && payload.quoteNumber);
    const mode = String((payload && payload.mode) || 'strikethrough');
    if (!baseQuoteNumber) return { success: false, message: '삭제할 견적번호가 없습니다.' };

    const prefixDept = String(baseQuoteNumber).trim().split(/\s+/)[0];
    let department = access.author.department;
    if (CONFIG.DEPARTMENT_ROOTS[prefixDept]) {
      department = prefixDept;
    } else if (access.isAdmin && payload && payload.department && CONFIG.DEPARTMENT_ROOTS[String(payload.department).trim()]) {
      department = String(payload.department).trim();
    }

    const rootFolderId = resolveRootFolderId({ authorDepartment: department });
    const system = getExistingYearSystem_(year, rootFolderId);
    if (!system) return { success: false, message: '삭제할 견적 대장을 찾을 수 없습니다.' };
    const sheet = system.ledgerSheet;
    const row = findLedgerRow_(sheet, baseQuoteNumber);
    if (row < 2) return { success: false, message: '삭제할 견적을 대장에서 찾을 수 없습니다.' };

    if (mode === 'permanent') {
      // 대장 행 삭제
      sheet.deleteRow(row);
      // 스냅샷 삭제
      const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
      if (snapshot && snapshot.getLastRow() >= 2) {
        const snapRows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 2).getValues();
        for (let i = snapRows.length - 1; i >= 0; i--) {
          if (String(snapRows[i][1] || '').trim() === baseQuoteNumber) {
            snapshot.deleteRow(i + 2);
          }
        }
      }
      // 로컬 파일 폴더 삭제 요청 (에이전트가 실제 폴더를 찾아 삭제)
      enqueueLocalDelete_(department, year, baseQuoteNumber);
      enqueueActivityLog_(access.author.email || access.author.name || '', '삭제', `견적번호: ${baseQuoteNumber} | 영구삭제 | 부서: ${department}`);
      return { success: true, message: '견적이 영구 삭제되었습니다. 다음 견적은 빈 번호로 발행됩니다.' };
    }

    // 라인삭제: 대장 행 전체에 검은 취소선 (견적번호는 유지)
    sheet.getRange(row, 1, 1, sheet.getLastColumn()).setFontLine('line-through');
    enqueueActivityLog_(access.author.email || access.author.name || '', '삭제', `견적번호: ${baseQuoteNumber} | 라인삭제(취소선) | 부서: ${department}`);
    return { success: true, message: '견적에 취소선이 표시되었습니다. 견적번호는 유지됩니다.' };
  } catch (err) {
    return { success: false, message: '견적 삭제 실패: ' + err.toString() };
  } finally {
    lock.releaseLock();
  }
}

function processQuote(quoteData, createDraft, customSubject, customBody, revisionOf, revisionYear, revisionDepartment, saveMode, overwriteTargetNumber) {
  const submittedNotes = String(quoteData && quoteData.details && quoteData.details.notes || '');
  if (submittedNotes.length > MAX_QUOTE_NOTES_LENGTH) {
    return {
      success: false,
      message: '비고는 ' + MAX_QUOTE_NOTES_LENGTH + '자까지 입력할 수 있습니다. (현재 ' + submittedNotes.length + '자)'
    };
  }

  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (err) {
    return { success: false, message: '다른 사용자가 저장 중입니다. 잠시 후 다시 시도해 주세요.' };
  }

  try {
    const details = quoteData.details;
    details.operationId = details.operationId || Utilities.getUuid();
    const requestedBaseQuoteNumber = baseQuoteNumber_(revisionOf);
    // 수정본은 원본을 조회했을 때의 부서(revisionDepartment)를 그대로 사용해야 한다.
    // 폼의 작성자 드롭다운에서 다른 사람을 선택했더라도, 폴더/대장 조회는 반드시 원본이 저장된
    // 부서를 기준으로 해야 로컬 에이전트가 같은 폴더에 파일을 저장할 수 있다.
    if (requestedBaseQuoteNumber && String(revisionDepartment || '').trim()) {
      details.authorDepartment = String(revisionDepartment).trim();
    }
    const rootFolderId = resolveRootFolderId(details);

    // 수정본은 원본을 조회했던 연도(revisionYear)를 그대로 사용해야 한다.
    // 견적일자를 오늘 날짜로 채워도 원본이 저장된 연도의 대장을 찾도록 한다.
    // 미래 연도·존재하지 않는 연도가 들어오면 대장/폴더를 새로 만들지 않고 즉시 실패시킨다.
    const nowYear = new Date().getFullYear();
    let currentYear;
    let system;
    const parsedRevisionYear = Number(revisionYear);
    if (requestedBaseQuoteNumber) {
      if (Number.isInteger(parsedRevisionYear) && parsedRevisionYear >= 2000 && parsedRevisionYear <= nowYear) {
        currentYear = parsedRevisionYear;
      } else {
        const dateMatches = String(details.quoteDate || '').match(/(\d{4})/);
        currentYear = dateMatches ? Math.min(parseInt(dateMatches[1], 10), nowYear) : nowYear;
      }
      // 수정 대상은 기존 연도 폴더/대장이 반드시 있어야 한다 — 없으면 새로 만들지 않고 실패 응답한다.
      system = getExistingYearSystem_(currentYear, rootFolderId);
      if (!system || findLedgerRow_(system.ledgerSheet, requestedBaseQuoteNumber) < 2) {
        lock.releaseLock();
        return { success: false, message: '수정할 원본 견적을 찾을 수 없습니다. 연도를 확인해 주세요.' };
      }
    } else {
      const dateMatches = String(details.quoteDate || '').match(/(\d{4})/);
      currentYear = dateMatches ? parseInt(dateMatches[1], 10) : nowYear;
      system = getYearSystem(currentYear, rootFolderId);
    }

    let realQuoteNumber;
    let baseQuoteNumber;
    let revisionNumber = 0;
    if (requestedBaseQuoteNumber) {
      baseQuoteNumber = requestedBaseQuoteNumber;
      if (saveMode === 'overwrite' && overwriteTargetNumber) {
        // 덮어쓰기 모드: 사용자가 선택한 기존 버전 번호(원본 또는 특정 Rev)를 그대로 사용
        realQuoteNumber = String(overwriteTargetNumber).trim();
        // 대상은 현재 수정 중인 기본 견적 또는 해당 기본 견적의 Rev만 허용한다.
        // 임의의 견적번호를 전달해 다른 견적 폴더를 덮어쓰는 것을 방지한다.
        const targetBaseNumber = baseQuoteNumber_(realQuoteNumber);
        if (targetBaseNumber !== baseQuoteNumber || !new RegExp('^' + baseQuoteNumber.replace(/[.*+?^${}()|[\\]\\]/g, '\\$&') + '(?:_Rev\\d+)?$', 'i').test(realQuoteNumber)) {
          lock.releaseLock();
          return { success: false, message: '대체 대상 견적서가 현재 수정 대상과 일치하지 않습니다.' };
        }
        const targetSnapshot = getQuoteSnapshotSheet_(system.ledger, false);
        let targetExists = realQuoteNumber === baseQuoteNumber;
        if (targetSnapshot && targetSnapshot.getLastRow() >= 2) {
          const targetRows = targetSnapshot.getRange(2, 1, targetSnapshot.getLastRow() - 1, 2).getDisplayValues();
          targetExists = targetExists || targetRows.some(function(row) {
            return String(row[0] || '').trim() === realQuoteNumber && String(row[1] || '').trim() === baseQuoteNumber;
          });
        }
        if (!targetExists) {
          lock.releaseLock();
          return { success: false, message: '대체할 기존 견적서 버전을 찾을 수 없습니다.' };
        }
        const revMatch = realQuoteNumber.match(/_Rev(\d+)$/i);
        revisionNumber = revMatch ? parseInt(revMatch[1], 10) : 0;
        details.overwrite = true;
      } else {
        // 다른이름으로 저장 (새 리비전 번호 발급)
        revisionNumber = getNextRevisionNumber_(system.ledger, system.ledgerSheet, baseQuoteNumber);
        realQuoteNumber = `${baseQuoteNumber}_Rev${String(revisionNumber).padStart(2, '0')}`;
      }
      details.baseQuoteNumber = baseQuoteNumber;
      details.revisionNumber = revisionNumber;
      details.folderClientName = getQuoteFolderClientName_(system.ledger, baseQuoteNumber) || details.clientName;
    } else {
      realQuoteNumber = getNextQuoteNumber(rootFolderId, details.authorDepartment);
      baseQuoteNumber = realQuoteNumber;
      details.folderClientName = details.clientName;
    }
    details.quoteNumber = realQuoteNumber;
    // 로컬 에이전트는 details.quoteDate(항상 오늘 날짜)가 아니라 이 값으로 저장 연도 폴더를 결정해야
    // 수정본을 원본과 같은 연도 폴더에 저장할 수 있다.
    details.year = currentYear;

    const items = quoteData.items || [];
    const itemParts = splitQuoteItems_(items);
    const finalVatPrice = Math.round(items.reduce(function(sum, item) {
      return sum + Number(item.totalPrice || 0);
    }, 0) * 1.1);

    // 파일 생성 작업을 먼저 대기열에 등록한다.
    // 대장 행은 에이전트가 XLSX/PDF 생성에 성공했다는 완료 보고를 보낸 뒤 processAgentResults에서 기록한다.
    details.agentQueuedAt = Date.now();
    enqueueLocalSave_(details, items, finalVatPrice, createDraft, customSubject);
    setQuoteOperationStatus_(details.operationId, 'pending', realQuoteNumber);

    let returnMessage = '[접수] 견적번호: ' + realQuoteNumber;
    if (itemParts.length > 1) {
      returnMessage += `\n${itemParts.length}개 파일로 분할 생성됩니다. (파일당 최대 ${ITEMS_PER_FILE}개 품목)`;
    }
    returnMessage += '\n파일 생성이 성공한 뒤 견적관리대장에 자동 기록됩니다.';

    if (createDraft) {
      if (details.clientEmail && details.clientEmail.indexOf('@') > -1) {
        // Gmail 초안 첨부용으로만 임시 PDF를 만들고, 초안 생성 직후 구글 드라이브에서 삭제한다.
        const tempPdfStartedAt = Date.now();
        const pdfBlobs = itemParts.map(function(partItems, partIndex) {
          const partDetails = Object.assign({}, details, {
            quoteNumber: quotePartNumber_(realQuoteNumber, partIndex, itemParts.length)
          });
          return createTempQuotePdf_({ details: partDetails, items: partItems }, system, partDetails);
        });
        Logger.log(`[성능] 메일 첨부 PDF 생성 ${realQuoteNumber}: ${Date.now() - tempPdfStartedAt}ms, ${pdfBlobs.length}개`);
        const emailBodyText = customBody || makeDefaultEmailBody(details);
        const htmlBody = textToHtmlBody_(emailBodyText);

        // HTML 모드로 초안을 생성하여 사용자의 Gmail 서명 이미지/로고가 깨지지 않도록 한다.
        const draftStartedAt = Date.now();
        GmailApp.createDraft(
          details.clientEmail,
          customSubject || `[CIMON] ${details.clientName} - 제품 견적서 송부 드립니다.`,
          emailBodyText,
          {
            attachments: pdfBlobs,
            name: details.authorName || 'CIMON',
            htmlBody: htmlBody
          }
        );
        Logger.log(`[성능] Gmail 초안 생성 ${realQuoteNumber}: ${Date.now() - draftStartedAt}ms`);
        returnMessage += '\n\nGmail 초안 생성 완료!';
      } else {
        returnMessage += '\n\n이메일 주소 오류: Gmail 초안 생성 실패';
      }
    }

    lock.releaseLock();
    return {
      success: true,
      url: '',
      pdfUrl: '',
      folderUrl: system.folder.getUrl(),
      sheetUrl: '',
      newQuoteNumber: realQuoteNumber,
      baseQuoteNumber: baseQuoteNumber,
      revisionNumber: revisionNumber || undefined,
      operationId: details.operationId,
      message: returnMessage
    };
  } catch (err) {
    lock.releaseLock();
    const errorText = String(err && err.toString ? err.toString() : err);
    if (/DriveApp|액세스가 거부|Access denied|권한이 없습니다/i.test(errorText)) {
      return {
        success: false,
        message: '견적 저장에 필요한 Google Drive 권한이 없습니다. 관리자에게 부서 폴더·대장 템플릿·견적에이전트 폴더의 편집 권한을 요청해 주세요. 관리자 계정에서 diagnoseUserDriveAccess를 실행하면 누락된 권한을 확인할 수 있습니다.'
      };
    }
    return { success: false, message: '작업 실패: ' + errorText };
  }
}

// Gmail 초안 첨부용 임시 PDF 생성. 시트 복사 → 채우기 → PDF 내보내기 → 원본 시트 즉시 삭제.
function createTempQuotePdf_(quoteData, system, details) {
  const tempName = `임시_${sanitizeDriveName(details.quoteNumber)}_${Date.now()}`;
  const templateFile = DriveApp.getFileById(CONFIG.TEMPLATE_ID);
  const newFile = templateFile.makeCopy(tempName, system.folder);
  try {
    const spreadsheet = SpreadsheetApp.open(newFile);
    const sheet = spreadsheet.getSheets()[0];
    const baseItemRowHeights = (quoteData.items || []).map(function(item, index) {
      return sheet.getRowHeight(CELL_MAP.itemStartRow + index);
    });
    fillSheetData(spreadsheet, quoteData);
    SpreadsheetApp.flush();
    const sheetId = sheet.getSheetId();
    return createPdfBlob(
      newFile.getId(),
      sheetId,
      `${sanitizeDriveName(details.quoteNumber)}_${sanitizeDriveName(details.clientName)}_견적서`,
      sheet,
      quoteData.items || [],
      baseItemRowHeights
    );
  } finally {
    permanentlyDeleteFile_(newFile.getId());
  }
}

// 드라이브 파일을 휴지통이 아닌 영구 삭제로 제거한다 (용량 즉시 회수).
// 공유 드라이브(Shared Drive) 항목 삭제에는 supportsAllDrives 파라미터가 필요하다.
function permanentlyDeleteFile_(fileId) {
  try {
    const res = UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/files/' + fileId + '?supportsAllDrives=true', {
      method: 'delete',
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      muteHttpExceptions: true
    });
    if (res.getResponseCode() !== 204) {
      Logger.log('파일 영구 삭제 실패(' + fileId + '): ' + res.getContentText());
    }
  } catch (err) {
    Logger.log('파일 영구 삭제 예외(' + fileId + '): ' + err.toString());
  }
}

// ── 사내 PC 로컬 저장 에이전트 (Google Drive 폴더 연동) 끝 ────────────────

function normalizeQuoteTemplateLabel_(value) {
  return String(value || '').replace(/[\s:：]/g, '').toLowerCase();
}

// 임시 견적서 템플릿의 라벨을 찾아 그 오른쪽 값 셀을 반환한다.
// 업체별/버전별 템플릿에서 작성자 영역의 행·열 위치가 달라도 라벨 기준으로 맞춘다.
function findQuoteTemplateValueCell_(sheet, label) {
  const firstSearchRow = 30; // 고객 정보 라벨과 혼동하지 않도록 하단 작성자 영역만 검색
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  if (lastRow < firstSearchRow || lastColumn < 1) return null;

  const labels = sheet.getRange(firstSearchRow, 1, lastRow - firstSearchRow + 1, lastColumn).getDisplayValues();
  const target = normalizeQuoteTemplateLabel_(label);
  for (let rowOffset = labels.length - 1; rowOffset >= 0; rowOffset -= 1) {
    for (let columnOffset = 0; columnOffset < labels[rowOffset].length; columnOffset += 1) {
      if (normalizeQuoteTemplateLabel_(labels[rowOffset][columnOffset]) !== target) continue;

      const row = firstSearchRow + rowOffset;
      const column = columnOffset + 1;
      let valueColumn = column + 1;
      try {
        const mergedRanges = sheet.getRange(row, column).getMergedRanges();
        mergedRanges.forEach(function(mergedRange) {
          if (row >= mergedRange.getRow() && row <= mergedRange.getLastRow() &&
              column >= mergedRange.getColumn() && column <= mergedRange.getLastColumn()) {
            valueColumn = Math.max(valueColumn, mergedRange.getLastColumn() + 1);
          }
        });
      } catch (err) { /* 병합 셀을 조회할 수 없으면 라벨 바로 오른쪽을 사용한다. */ }

      if (valueColumn <= sheet.getMaxColumns()) return sheet.getRange(row, valueColumn);
    }
  }
  return null;
}

function setQuoteTemplateValueByLabel_(sheet, label, fallbackA1, value) {
  const target = findQuoteTemplateValueCell_(sheet, label);
  if (target) {
    target.setValue(value || '');
  } else {
    // 라벨이 없는 구형 템플릿은 기존 H37:H39 위치로 호환한다.
    setRangeValuesIfRangeExists(sheet, fallbackA1, [[value || '']]);
  }
}

function mergedSheetCellWidthPx_(sheet, row, column) {
  const cell = sheet.getRange(row, column);
  const merges = cell.getMergedRanges();
  let startColumn = column;
  let endColumn = column;
  for (const merged of merges) {
    if (row >= merged.getRow() && row <= merged.getLastRow() &&
        column >= merged.getColumn() && column <= merged.getLastColumn()) {
      startColumn = merged.getColumn();
      endColumn = merged.getLastColumn();
      break;
    }
  }

  let width = 0;
  for (let col = startColumn; col <= endColumn; col += 1) width += sheet.getColumnWidth(col);
  return Math.max(width, 24);
}

function quoteTextDisplayUnits_(value) {
  return Array.from(String(value || '')).reduce(function(total, character) {
    return total + (/^[\u1100-\u11FF\u2E80-\uA4CF\uAC00-\uD7AF\uF900-\uFAFF\uFE10-\uFE6F\uFF00-\uFFEF]$/u.test(character) ? 2 : 1);
  }, 0);
}

function estimateQuoteCellLines_(value, widthPx, fontSizePt) {
  const fontSize = Math.max(Number(fontSizePt) || 10, 8);
  const averageCharacterPx = fontSize * 96 / 72 * 0.53;
  const charactersPerLine = Math.max(8, Math.floor(Math.max(averageCharacterPx * 8, widthPx - 12) / averageCharacterPx));
  return String(value || '').replace(/\r\n?/g, '\n').split('\n').reduce(function(total, line) {
    return total + Math.max(1, Math.ceil(quoteTextDisplayUnits_(line) / charactersPerLine));
  }, 0);
}

function sizeQuotePdfItemRows_(sheet, items, firstRow) {
  if (!items || items.length === 0) return;

  sheet.getRange(firstRow, 2, items.length, 1).setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);
  sheet.getRange(firstRow, 4, items.length, 1).setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);

  items.forEach(function(item, index) {
    const row = firstRow + index;
    const nameCell = sheet.getRange(row, 2);
    const specCell = sheet.getRange(row, 4);
    const nameFontSize = Number(nameCell.getFontSize()) || 10;
    const specFontSize = Number(specCell.getFontSize()) || 10;
    const nameWidth = mergedSheetCellWidthPx_(sheet, row, 2);
    const specWidth = mergedSheetCellWidthPx_(sheet, row, 4);
    let fontSize = Math.max(8, Math.floor(Math.min(nameFontSize, specFontSize, 10)));
    let estimatedHeightPx = 0;

    for (; fontSize >= 8; fontSize -= 1) {
      const lineCount = Math.max(
        estimateQuoteCellLines_(item.name, nameWidth, fontSize),
        estimateQuoteCellLines_(item.spec, specWidth, fontSize)
      );
      estimatedHeightPx = Math.ceil(lineCount * fontSize * 96 / 72 * 1.35 + 8);
      if (estimatedHeightPx <= MAX_QUOTE_PDF_ITEM_ROW_HEIGHT_PX || fontSize === 8) break;
    }

    nameCell.setFontSize(fontSize);
    specCell.setFontSize(fontSize);
    // 8pt에서도 상한을 넘는 극단 길이에서는 텍스트를 자르지 않고 필요한 높이를 허용한다.
    sheet.setRowHeight(row, Math.max(sheet.getRowHeight(row), estimatedHeightPx));
  });
}

function sizeQuotePdfNotes_(sheet, notes) {
  const noteRange = sheet.getRange('B34:F40');
  noteRange.setWrap(true);
  noteRange.setVerticalAlignment('top');

  const noteCell = sheet.getRange(34, 2);
  const widthPx = mergedSheetCellWidthPx_(sheet, 34, 2);
  let availableHeightPx = 0;
  for (let row = 34; row <= 40; row += 1) availableHeightPx += sheet.getRowHeight(row);

  let fontSize = Math.min(Math.floor(Number(noteCell.getFontSize()) || 9), 9);
  while (fontSize > 7) {
    const lines = estimateQuoteCellLines_(notes, widthPx, fontSize);
    if (lines * fontSize * 96 / 72 * 1.25 + 6 <= availableHeightPx) break;
    fontSize -= 1;
  }
  noteRange.setFontSize(Math.max(7, fontSize));
}

function compactQuotePdfLayoutForRetry_(sheet, items, firstRow, baseRowHeights) {
  items.forEach(function(item, index) {
    const row = firstRow + index;
    const nameCell = sheet.getRange(row, 2);
    const specCell = sheet.getRange(row, 4);
    nameCell.setFontSize(8);
    specCell.setFontSize(8);

    const lineCount = Math.max(
      estimateQuoteCellLines_(item.name, mergedSheetCellWidthPx_(sheet, row, 2), 8),
      estimateQuoteCellLines_(item.spec, mergedSheetCellWidthPx_(sheet, row, 4), 8)
    );
    const height = Math.ceil(lineCount * 8 * 96 / 72 * 1.2 + 6);
    sheet.setRowHeight(row, Math.max(Number(baseRowHeights[index]) || 36, height));
  });

  const noteRange = sheet.getRange('B34:F40');
  noteRange.setWrap(true);
  noteRange.setVerticalAlignment('top');
  noteRange.setFontSize(7);
}

function fillSheetData(spreadsheet, quoteData) {
  const sheet = spreadsheet.getSheets()[0];
  const details = quoteData.details || {};
  const items = quoteData.items || [];
  const notes = String(details.notes || '');
  if (notes.length > MAX_QUOTE_NOTES_LENGTH) {
    throw new Error('비고는 ' + MAX_QUOTE_NOTES_LENGTH + '자까지 입력할 수 있습니다. (현재 ' + notes.length + '자)');
  }

  // PDF 메일 첨부용 임시 시트는 기존에 셀마다 setValue를 반복했다.
  // Google Sheets 서비스 호출 수를 줄이기 위해 연속된 셀은 한 번에 기록한다.
  setRangeValuesIfRangeExists(sheet, 'C3:C4', [
    [details.clientName || ''],
    [details.clientContactPerson || ''],
  ]);
  setRangeValuesIfRangeExists(sheet, 'F3:F4', [
    [details.clientPhone || ''],
    [details.clientEmail || ''],
  ]);
  setRangeValuesIfRangeExists(sheet, 'I3:I4', [
    [details.quoteNumber || ''],
    [details.quoteDate || ''],
  ]);
  setRangeValuesIfRangeExists(sheet, 'C9:C13', [
    [details.deliveryLocation || ''],
    [details.deliveryDeadline || ''],
    [details.paymentTerms || ''],
    [details.validityPeriod || ''],
    [details.packing || ''],
  ]);
  // 작성자·연락처·이메일은 고정 셀 주소 대신 템플릿 라벨 옆 셀에 기록한다.
  setQuoteTemplateValueByLabel_(sheet, '작성자', 'H37', details.authorName || '');
  setQuoteTemplateValueByLabel_(sheet, '연락처', 'H38', details.authorPhone || '');
  setQuoteTemplateValueByLabel_(sheet, '이메일', 'H39', details.authorEmail || '');
  setRangeValuesIfRangeExists(sheet, CELL_MAP.notes, [[notes]]);
  sizeQuotePdfNotes_(sheet, notes);

  let supplyTotal = 0;
  const itemNumbers = [];
  const itemNames = [];
  const itemSpecs = [];
  const quantities = [];
  const effectiveUnitPrices = [];
  const itemTotals = [];
  for (let i = 0; i < items.length; i++) {
    const item = items[i];
    const quantity = Number(item.quantity || 0);
    const unitPrice = Number(item.unitPrice || 0);
    const multiplierValue = Number(item.multiplier || 1);
    const multiplier = isFinite(multiplierValue) && multiplierValue > 0 ? multiplierValue : 1;
    const effectiveUnitPrice = unitPrice * multiplier;
    const totalPrice = Math.round(quantity * effectiveUnitPrice);

    itemNumbers.push([i + 1]);
    itemNames.push([item.name || '']);
    itemSpecs.push([item.spec || '']);
    quantities.push([quantity]);
    effectiveUnitPrices.push([effectiveUnitPrice]);
    itemTotals.push([totalPrice]);

    supplyTotal += totalPrice;
  }

  if (items.length > 0) {
    const firstRow = CELL_MAP.itemStartRow;
    const lastRow = firstRow + items.length - 1;
    setRangeValuesIfRangeExists(sheet, `A${firstRow}:A${lastRow}`, itemNumbers);
    setRangeValuesIfRangeExists(sheet, `B${firstRow}:B${lastRow}`, itemNames);
    setRangeValuesIfRangeExists(sheet, `D${firstRow}:D${lastRow}`, itemSpecs);
    setRangeValuesIfRangeExists(sheet, `G${firstRow}:G${lastRow}`, quantities);
    // 템플릿에는 배율 열이 없으므로 단가 칸에 배율을 반영한 유효 단가를 기록한다.
    setRangeValuesIfRangeExists(sheet, `H${firstRow}:H${lastRow}`, effectiveUnitPrices);
    setRangeValuesIfRangeExists(sheet, `I${firstRow}:I${lastRow}`, itemTotals);

    try {
      sizeQuotePdfItemRows_(sheet, items, firstRow);
    } catch (err) {
      Logger.log('메일 첨부 PDF 규격 행 높이 자동 조정 실패: ' + err.toString());
      // 템플릿 구조가 달라도 최소한 이름/규격 셀 줄바꿈은 유지한다.
      try {
        sheet.getRange(`B${firstRow}:B${lastRow}`).setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);
        sheet.getRange(`D${firstRow}:D${lastRow}`).setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);
      } catch (wrapErr) { /* 템플릿 차이가 큰 경우에도 견적 데이터 기록은 계속한다. */ }
    }
  }

  const vatIncludedPrice = Math.round(supplyTotal * 1.1);
  const formattedPrice = vatIncludedPrice.toLocaleString('ko-KR');
  setRangeValuesIfRangeExists(sheet, CELL_MAP.totalPriceCell, [[`총 견적 금액(VAT포함): ${formattedPrice} 원`]]);

  return vatIncludedPrice;
}

function quotePdfPageCount_(blob) {
  const bytes = blob.getBytes();
  let ascii = '';
  for (let i = 0; i < bytes.length; i += 1) ascii += String.fromCharCode(bytes[i] & 255);
  const pages = ascii.match(/\/Type\s*\/Page\b/g);
  return pages ? pages.length : 0;
}

function fetchQuotePdfBlob_(spreadsheetId, sheetId, fileName) {
  // scale=4는 Google Sheets PDF 내보내기의 "페이지에 맞춤"이며, 범위는 A1:J42로 제한한다.
  const url = 'https://docs.google.com/spreadsheets/d/' + spreadsheetId +
    '/export?format=pdf&size=A4&portrait=true&scale=4&gridlines=false&sheetnames=false' +
    '&printtitle=false&pagenum=UNDEFINED&attachment=true&gid=' + sheetId +
    '&r1=0&c1=0&r2=42&c2=10';
  const params = {
    method: 'GET',
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
    muteHttpExceptions: true
  };

  const response = UrlFetchApp.fetch(url, params);
  if (response.getResponseCode() !== 200) {
    throw new Error('메일 첨부 PDF 내보내기 실패: HTTP ' + response.getResponseCode());
  }
  return response.getBlob().setName(fileName + '.pdf');
}

function createPdfBlob(spreadsheetId, sheetId, fileName, sheet, items, baseRowHeights) {
  let blob = fetchQuotePdfBlob_(spreadsheetId, sheetId, fileName);
  let pageCount = quotePdfPageCount_(blob);
  if (pageCount === 1) return blob;

  if (pageCount > 1) {
    Logger.log('페이지 맞춤 PDF가 ' + pageCount + '쪽입니다. 8pt/7pt 압축 설정으로 한 번 더 내보냅니다.');
    compactQuotePdfLayoutForRetry_(sheet, items, CELL_MAP.itemStartRow, baseRowHeights);
    SpreadsheetApp.flush();
    blob = fetchQuotePdfBlob_(spreadsheetId, sheetId, fileName);
    pageCount = quotePdfPageCount_(blob);
    if (pageCount === 1) return blob;
  }

  if (pageCount !== 1) {
    throw new Error('메일 첨부 PDF가 1장으로 맞춰지지 않았습니다. 내보낸 페이지 수: ' + (pageCount || '확인 불가'));
  }
  return blob;
}

function saveToLedgerSheet(sheet, details, items, totalPrice, fileUrl) {
  // 신규/수정 행을 고정 열 번호(LEDGER_COL)에 그대로 쓰기 전에 실제 헤더가 예상과 일치하는지
  // 검증한다 — 대장 템플릿의 열 구성이 다르면 조용히 잘못된 열에 쓰는 대신 즉시 실패시킨다.
  assertLedgerLayout_(sheet);
  const quoteDate = String(details.quoteDate || '');
  const dateMatches = quoteDate.match(/(\d{4}).*?(\d{1,2}).*?(\d{1,2})/);
  const year = dateMatches ? dateMatches[1] : new Date().getFullYear();
  const month = dateMatches ? dateMatches[2] : new Date().getMonth() + 1;
  const day = dateMatches ? dateMatches[3] : new Date().getDate();
  // 제품명 요약: 7건 이하이면 모두 표시, 7건 초과이면 앞의 7건 + '외 N건'으로 표시
  const itemNames = items.map(function(item) { return String(item.name || '').trim(); }).filter(Boolean);
  let productName = '';
  if (itemNames.length <= 7) {
    productName = itemNames.join(', ');
  } else {
    const top7 = itemNames.slice(0, 7).join(', ');
    const remaining = itemNames.length - 7;
    productName = `${top7} 외 ${remaining}건`;
  }
  const categoryRules = [
    ['SCADA PRO', /SCADA\s*PRO/i],
    ['SCADA', /SCADA/i],
    ['TOUCH MONITOR', /TOUCH\s*MONITOR/i],
    ['BOX PC', /BOX\s*PC|\bNB\d/i],
    ['Hybird', /HYBRID|HYBIRD/i],
    ['Accessory', /ACCESSORY|ACCESSARY|액세서리/i],
    ['eXT', /\beXT\d*\b/i],
    ['XPANEL', /XPANEL/i],
    ['PLC', /\bPLC\b|\bCM[013]\b|NET\/RIO|CIMON-NET|REMOTE\s*IO|\bRIO\b/i],
    ['TOUCH', /TOUCH|50000_70000|5000SERIES|500SERIES|\biN[TP]\d/i]
  ];
  const productCategories = [];
  items.forEach(function(item) {
    const source = String(item.type || '') + ' ' + String(item.name || '');
    const matched = categoryRules.find(function(rule) { return rule[1].test(source); });
    if (matched && productCategories.indexOf(matched[0]) === -1) productCategories.push(matched[0]);
  });

  // 수정본은 원본 행을 갱신하고, 원본 견적번호는 대장에 그대로 유지한다.
  const ledgerQuoteNumber = String(details.baseQuoteNumber || details.quoteNumber || '').trim();
  // 대장 NO는 기본 견적번호의 일련번호(-NNN)와 항상 일치시킨다.
  // 견적번호 파싱에 실패한 경우에만 행 개수로 보정한다.
  const seqMatch = ledgerQuoteNumber.match(/-(\d+)\s*$/);
  const no = seqMatch ? parseInt(seqMatch[1], 10) : sheet.getLastRow();

  // G열(견적번호)부터 O열(비고)까지 9개 값 — E열(발주), P열(파일링크),
  // Q열(적용현장)은 별도 관리하므로 여기서 덮어쓰지 않는다.
  const quoteInfoValues = [
    ledgerQuoteNumber,
    details.clientName,
    details.clientContactPerson,
    details.clientPhone,
    details.clientEmail,
    productCategories.join(', '),
    productName,
    totalPrice,
    details.notes || ''
  ];

  // 신규 견적도 완료 보고 재처리 시 같은 견적번호 행을 중복 append하지 않도록 먼저 찾는다.
  const existingRow = findLedgerRow_(sheet, ledgerQuoteNumber);
  if (details.baseQuoteNumber && existingRow < 2) {
    throw new Error('수정 대상 원본 행을 찾을 수 없습니다.');
  }

  // 스냅샷 저장을 먼저 시도한다 — 스냅샷이 실패하면(예: 셀 글자수 한도) 대장 행은 아직
  // 손대지 않은 상태이므로 "일부만 반영된" 상태가 되지 않는다.
  saveQuoteSnapshot_(sheet.getParent(), details, items, totalPrice);

  if (existingRow >= 2) {
    // 수정본: A~E열(날짜/발주)은 원본 값을 보존하고, F열 작성자와 G~O열 견적 정보를 갱신한다.
    // P열(파일링크)과 Q열(적용현장)은 여기서 건드리지 않는다.
    if (String(details.authorName || '').trim()) {
      sheet.getRange(existingRow, LEDGER_COL.AUTHOR).setValue(String(details.authorName).trim());
    }
    sheet.getRange(existingRow, LEDGER_COL.QUOTE_NUMBER, 1, quoteInfoValues.length).setValues([quoteInfoValues]);
  } else {
    // 신규 견적: A~D 날짜, E 발주 공란, F 작성자, G~O 견적 정보, P 링크, Q 현장을 기록한다.
    sheet.appendRow([no, year, month, day, '', String(details.authorName || '').trim()]
      .concat(quoteInfoValues)
      .concat([fileUrl, '']));
  }
}

function generateAIContent(prompt) {
  if (!CONFIG.GEMINI_API_KEY) {
    return 'Gemini API key가 설정되지 않았습니다.';
  }

  const url = CONFIG.GEMINI_URL_BASE + CONFIG.GEMINI_API_KEY;
  const payload = { contents: [{ parts: [{ text: prompt }] }] };

  try {
    const response = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });
    const result = JSON.parse(response.getContentText());
    return result.candidates && result.candidates[0] && result.candidates[0].content && result.candidates[0].content.parts
      ? result.candidates[0].content.parts[0].text.trim()
      : '응답 없음';
  } catch (err) {
    return '에러';
  }
}

// 일반 텍스트를 HTML 본문으로 변환 (줄바꿈 및 특수문자 변환)
// options.htmlBody에 전달하여 Gmail 임시보관함 초안이 HTML 모드로 생성되도록 한다 (서명 이미지/로고 보존).
function textToHtmlBody_(text) {
  if (!text) return '';
  const escaped = String(text)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
  return escaped.replace(/\r\n|\r|\n/g, '<br>\n');
}

// ── 파일링크 생성 헬퍼 (공용) ───────────────────────────────────────────────
// 견적 번호/업체명 → 대장 P열(파일링크)에 기록할 서명된 링크 목록을 만든다.

function signedFileUrl_(dept, year, folderName, fileName) {
  const publicBaseUrl = 'http://172.35.12.36:8790';
  const fileLinkSecret = 'CIMON-FILES-2026-jR4wN8sQ2tVx7LmE';
  const relPath = [dept, String(year), folderName, fileName].map(encodeURIComponent).join('/');
  // 한글(멀티바이트)이 들어간 경로를 그대로 서명하면 Apps Script와 Node가 문자열→바이트 처리가
  // 달라 서명이 어긋난다. URL 인코딩된 ASCII 경로로 서명하면 양쪽 HMAC이 정확히 일치한다.
  const sigBytes = Utilities.computeHmacSha256Signature(relPath, fileLinkSecret);
  const sig = sigBytes.map(function(b) {
    const v = (b < 0 ? b + 256 : b).toString(16);
    return v.length === 1 ? '0' + v : v;
  }).join('');
  return `${publicBaseUrl}/files/${relPath}?k=${sig}`;
}

function buildRowLinkItems_(dept, year, quoteNumber, company, snapshotRows) {
  const baseQuote = baseQuoteNumber_(quoteNumber);
  const matchedSnapshots = snapshotRows.filter(function(row) {
    return String(row[1] || '').trim() === baseQuote;
  });

  // 실제 폴더명에 사용된 업체명(스냅샷의 folderClientName)을 우선 사용한다.
  let folderClientName = company;
  for (let si = matchedSnapshots.length - 1; si >= 0; si--) {
    try {
      const stored = JSON.parse(String(matchedSnapshots[si][4] || '{}'));
      const storedClient = String((stored.details && (stored.details.folderClientName || stored.details.clientName)) || '').trim();
      if (storedClient) {
        folderClientName = storedClient;
        break;
      }
    } catch (err) { /* 다음 스냅샷 행으로 넘어간다 */ }
  }

  const folderName = `${baseQuote}_${folderClientName}`;
  const linkItems = [];
  const seenUrls = {};

  const origFileName = `${baseQuote}_${folderClientName}_견적서.pdf`;
  const origUrl = signedFileUrl_(dept, year, folderName, origFileName);
  linkItems.push({ text: `원본: ${origFileName}`, url: origUrl });
  seenUrls[origUrl] = true;

  matchedSnapshots.forEach(function(row) {
    const outNum = String(row[0] || '').trim();
    const revMatch = outNum.match(/_Rev(\d+)$/i);
    if (!revMatch) return;
    const revNum = parseInt(revMatch[1], 10);
    const revFileName = `${outNum}_${folderClientName}_견적서.pdf`;
    const revUrl = signedFileUrl_(dept, year, folderName, revFileName);
    if (seenUrls[revUrl]) return;
    seenUrls[revUrl] = true;
    linkItems.push({ text: `Rev${String(revNum).padStart(2, '0')}: ${revFileName}`, url: revUrl });
  });

  linkItems.sort(function(a, b) {
    const aRev = a.text.match(/^Rev(\d+)/);
    const bRev = b.text.match(/^Rev(\d+)/);
    const aNum = aRev ? parseInt(aRev[1], 10) : 0;
    const bNum = bRev ? parseInt(bRev[1], 10) : 0;
    return aNum - bNum;
  });
  return linkItems;
}

function writeLinkItemsToCell_(sheet, rowIndex, linkItems) {
  if (!linkItems || linkItems.length === 0) return;
  const separator = ' | ';
  const combinedText = linkItems.map(function(item) { return item.text; }).join(separator);
  const builder = SpreadsheetApp.newRichTextValue().setText(combinedText);
  let offset = 0;
  linkItems.forEach(function(item) {
    if (item.url) builder.setLinkUrl(offset, offset + item.text.length, item.url);
    offset += item.text.length + separator.length;
  });
  sheet.getRange(rowIndex, LEDGER_COL.FILE_LINK).setRichTextValue(builder.build());
}

function isLegacyFileLinkInNotes_(value) {
  const text = String(value || '').trim();
  if (!text) return false;
  return /^(?:원본|Rev\d+):\s*[^|]*_견적서\.pdf(?:\s*\|\s*(?:원본|Rev\d+):\s*[^|]*_견적서\.pdf)*$/i.test(text);
}

function latestSnapshotNotes_(snapshotRows, quoteNumber) {
  const baseQuote = baseQuoteNumber_(quoteNumber);
  for (let index = snapshotRows.length - 1; index >= 0; index -= 1) {
    const snapshotRow = snapshotRows[index];
    const q0 = String(snapshotRow[0] || '').trim();
    const q1 = String(snapshotRow[1] || '').trim();
    if (q1 !== baseQuote && q0 !== baseQuote && baseQuoteNumber_(q0) !== baseQuote) continue;
    try {
      const stored = JSON.parse(String(snapshotRow[4] || snapshotRow[3] || '{}'));
      if (stored.details && stored.details.notes !== undefined) {
        return { found: true, value: String(stored.details.notes || '') };
      }
    } catch (err) { /* 손상된 스냅샷은 더 이전 기록에서 복구를 시도한다. */ }
  }
  return { found: false, value: '' };
}

function getAgentStatsNotesMap_(department) {
  const map = {};
  try {
    const statsFolder = getAgentSubfolder_('stats');
    const files = statsFolder.getFilesByName(String(department || '') + '.json');
    if (!files.hasNext()) return map;
    const stats = JSON.parse(files.next().getBlob().getDataAsString('utf8'));
    const records = Array.isArray(stats.records) ? stats.records : [];
    for (let index = 0; index < records.length; index += 1) {
      const record = records[index];
      const recordQuote = String(record.quoteNumber || '').trim();
      if (recordQuote && record.notes !== undefined) {
        map[recordQuote] = String(record.notes || '');
        const base = baseQuoteNumber_(recordQuote);
        if (base && !map[base]) map[base] = String(record.notes || '');
      }
    }
  } catch (err) {
    Logger.log('부서 통계 비고 맵 생성 예외: ' + err.toString());
  }
  return map;
}

// 견적 목록 조회 시 누락 링크를 복구하고, 예전 열 매핑이 P열 링크를 O열 비고에 썼던 흔적도 복구한다.
function repairEmptyLedgerLinks_(system, dept, year, rows) {
  const linkCol = LEDGER_COL.FILE_LINK - 1;
  const quoteCol = LEDGER_COL.QUOTE_NUMBER - 1;
  const companyCol = LEDGER_COL.CLIENT_NAME - 1;
  const notesCol = LEDGER_COL.NOTES - 1;

  // 빈 링크와 레거시 파일명 텍스트가 비고에 남아 있는 행만 처리한다.
  let hasMissing = false;
  let hasLegacyNotes = false;
  for (let i = 0; i < rows.length; i++) {
    if (!rows[i].links[linkCol]) hasMissing = true;
    if (isLegacyFileLinkInNotes_(rows[i].values[notesCol])) {
      hasMissing = true;
      hasLegacyNotes = true;
    }
    if (hasMissing && hasLegacyNotes) break;
  }
  if (!hasMissing) return 0;

  const sheet = system.ledgerSheet;
  const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
  const snapshotRows = (snapshot && snapshot.getLastRow() >= 2)
    ? snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues()
    : [];

  // 통계 파일은 레거시 비고 복구가 필요한 행이 있고 스냅샷에서 찾지 못했을 때만 최대 1회 읽는다.
  let statsNotesMap = null;
  const startTime = Date.now();
  let repaired = 0;

  for (let i = 0; i < rows.length; i++) {
    // 웹 응답 타임아웃(30초)을 절대 넘기지 않도록 복구 작업은 10초로 제한한다
    if (Date.now() - startTime > 10000) {
      Logger.log('[성능] repairEmptyLedgerLinks_ 10초 상한 도달, 남은 행은 다음 조회에 이어서 처리합니다.');
      break;
    }

    const row = rows[i];
    const quoteNumber = String(row.values[quoteCol] || '').trim();
    const company = String(row.values[companyCol] || '').trim();
    if (!quoteNumber || !company) continue;

    if (!row.links[linkCol]) {
      const linkItems = buildRowLinkItems_(dept, year, quoteNumber, company, snapshotRows);
      if (linkItems.length) {
        writeLinkItemsToCell_(sheet, i + 2, linkItems);
        row.links[linkCol] = linkItems[0].url; // 화면에 즉시 표시되도록 반환 데이터에 반영
        repaired++;
      }
    }

    if (isLegacyFileLinkInNotes_(row.values[notesCol])) {
      let restoredNotes = latestSnapshotNotes_(snapshotRows, quoteNumber);
      if (!restoredNotes.found) {
        if (statsNotesMap === null) {
          statsNotesMap = getAgentStatsNotesMap_(dept);
        }
        const fromStats = statsNotesMap[quoteNumber] || statsNotesMap[baseQuoteNumber_(quoteNumber)];
        if (fromStats !== undefined) {
          restoredNotes = { found: true, value: fromStats };
        } else {
          // 스냅샷과 통계 어디에도 없는 경우: 파일명 텍스트는 버그로 들어간 가짜 비고이므로 빈 문자열로 복구한다.
          restoredNotes = { found: true, value: '' };
        }
      }
      if (restoredNotes.found) {
        sheet.getRange(i + 2, LEDGER_COL.NOTES).setValue(restoredNotes.value);
        row.values[notesCol] = restoredNotes.value;
        repaired++;
      }
    }
  }
  if (repaired > 0) SpreadsheetApp.flush();
  return repaired;
}

// ── 파일링크 복구 유틸리티 ──────────────────────────────────────────────────
// 에이전트 큐 삭제나 동기화 지연 등으로 대장 파일링크가 누락된 모든 견적 행의 링크를
// 스냅샷 기록을 바탕으로 자동 계산하여 대장에 복구한다.
// 사용 방법: 편집기 상단 함수 드롭다운에서 autoRepairAllMissingLedgerLinks 선택 → [실행] 클릭
function autoRepairAllMissingLedgerLinks() {
  const currentYear = new Date().getFullYear();
  for (const dept of Object.keys(CONFIG.DEPARTMENT_ROOTS)) {
    const rootFolderId = CONFIG.DEPARTMENT_ROOTS[dept];
    const system = getExistingYearSystem_(currentYear, rootFolderId);
    if (!system) continue;

    const sheet = system.ledgerSheet;
    const lastRow = sheet.getLastRow();
    if (lastRow < 2) continue;

    const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
    const snapshotRows = (snapshot && snapshot.getLastRow() >= 2)
      ? snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues()
      : [];

    let repaired = 0;
    for (let r = 2; r <= lastRow; r++) {
      const quoteNumber = String(sheet.getRange(r, LEDGER_COL.QUOTE_NUMBER).getDisplayValue() || '').trim();
      const company = String(sheet.getRange(r, LEDGER_COL.CLIENT_NAME).getDisplayValue() || '').trim();
      if (!quoteNumber || !company) continue;
      const linkItems = buildRowLinkItems_(dept, currentYear, quoteNumber, company, snapshotRows);
      if (!linkItems.length) continue;
      writeLinkItemsToCell_(sheet, r, linkItems);
      repaired++;
    }
    SpreadsheetApp.flush();
    Logger.log(`[부서: ${dept}] 파일링크 복구 완료: ${repaired}행`);
  }
  Logger.log('전체 부서 파일링크 복구 작업 완료!');
}

function repairMissingLedgerLink() {
  autoRepairAllMissingLedgerLinks();
}

// ── 전체 테스트 견적 초기화 유틸리티 (정식 오픈 전 1회 실행) ────────────────
// 3개 부서(기술영업, 영업, 프로젝트)의 대장 및 스냅샷 시트 데이터를 깨끗이 비우고,
// 큐 폴더의 잔여 파일을 정리하여 다음 견적이 001번부터 새로 시작하도록 초기화한다.
// 실행 방법: Apps Script 편집기 상단 드롭다운에서 resetAllTestQuotes 선택 → [실행] 클릭
function resetAllTestQuotes() {
  const currentYear = new Date().getFullYear();

  // 1. 모든 부서의 대장 및 스냅샷 시트 행 비우기 (1행 헤더는 안전하게 보존)
  for (const dept of Object.keys(CONFIG.DEPARTMENT_ROOTS)) {
    PropertiesService.getScriptProperties().deleteProperty(`QUOTE_SEQ_${sanitizeDriveName(dept)}_${currentYear}`);
    const rootFolderId = CONFIG.DEPARTMENT_ROOTS[dept];
    const system = getExistingYearSystem_(currentYear, rootFolderId);
    if (!system) {
      Logger.log(`[부서: ${dept}] ${currentYear}년 대장이 없어 건너뜁니다.`);
      continue;
    }

    // 1-1. 대장 시트(견적관리대장) 행 비우기 (2행부터 마지막 행까지 삭제)
    const ledgerSheet = system.ledgerSheet;
    const ledgerLastRow = ledgerSheet.getLastRow();
    if (ledgerLastRow >= 2) {
      ledgerSheet.deleteRows(2, ledgerLastRow - 1);
      Logger.log(`[부서: ${dept}] 견적관리대장 ${ledgerLastRow - 1}개 행 삭제 완료`);
    } else {
      Logger.log(`[부서: ${dept}] 견적관리대장이 이미 비어 있습니다.`);
    }

    // 1-2. 스냅샷 시트(견적원본데이터) 행 비우기 (2행부터 마지막 행까지 삭제)
    const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
    if (snapshot) {
      const snapLastRow = snapshot.getLastRow();
      if (snapLastRow >= 2) {
        snapshot.deleteRows(2, snapLastRow - 1);
        Logger.log(`[부서: ${dept}] 견적원본데이터(스냅샷) ${snapLastRow - 1}개 행 삭제 완료`);
      } else {
        Logger.log(`[부서: ${dept}] 견적원본데이터 시트가 이미 비어 있습니다.`);
      }
    }
    SpreadsheetApp.flush();
  }

  // 2. 에이전트 큐 폴더 안의 잔여 파일들만 삭제 (폴더 자체는 보존!)
  try {
    ['pending', 'results', 'delivery', 'failed', 'resultsFailed'].forEach(function(subName) {
      try {
        const subFolder = getAgentSubfolder_(subName);
        const files = subFolder.getFiles();
        let deletedCount = 0;
        while (files.hasNext()) {
          const file = files.next();
          permanentlyDeleteFile_(file.getId());
          deletedCount++;
        }
        Logger.log(`[큐 폴더: ${subName}] 잔여 파일 ${deletedCount}개 정리 완료`);
      } catch (err) {
        Logger.log(`[큐 폴더: ${subName}] 정리 실패: ${err.toString()}`);
      }
    });
  } catch (agentErr) {
    Logger.log(`에이전트 큐 정리 오류: ${agentErr.toString()}`);
  }

  Logger.log('====================================================');
  Logger.log('전체 부서(기술영업/영업/프로젝트) 초기화가 완료되었습니다!');
  Logger.log('이제 각 부서의 첫 견적서는 001번부터 정상 채번됩니다.');
  Logger.log('====================================================');
}

// ── 파일링크 서명 진단 (일회용) ──────────────────────────────────────────────
// Apps Script 편집기에서 diagnoseFileLinkHmac 선택 → 실행 → 로그를 보면
// Apps Script가 내는 서명과 Node(에이전트)가 내는 서명이 일치하는지 확인할 수 있다.
// 실행 후 왼쪽 "실행(Executions)"에서 로그 확인.
function diagnoseFileLinkHmac() {
  const secret = 'CIMON-FILES-2026-jR4wN8sQ2tVx7LmE';

  const url = signedFileUrl_('기술영업', 2026, '기술영업 2609-001_test1', '기술영업 2609-001_test1_견적서.pdf');
  Logger.log('생성 URL: ' + url);

  // URL 인코딩된 ASCII 경로로 서명한 값 (에이전트가 검증하는 방식과 동일)
  const relPath = ['기술영업', '2026', '기술영업 2609-001_test1', '기술영업 2609-001_test1_견적서.pdf']
    .map(encodeURIComponent).join('/');
  const bytes = Utilities.computeHmacSha256Signature(relPath, secret);
  const hex = bytes.map(function(b) {
    const v = (b < 0 ? b + 256 : b).toString(16);
    return v.length === 1 ? '0' + v : v;
  }).join('');
  Logger.log('Apps Script HMAC(인코딩 경로, hex): ' + hex);
  Logger.log('이 값이 에이전트가 URL로 검증하는 서명과 일치해야 합니다.');
}

// ── 대장만 있고 로컬 파일이 없는 견적 재처리 유틸리티 ────────────────────────
// Apps Script가 대장은 저장했지만 pending 큐 기록/Drive 동기화가 실패한 경우,
// 견적원본데이터 스냅샷에서 원본/리비전 작업을 다시 만들어 로컬 에이전트에 전달한다.
// 실행 방법: 함수 선택 → requeueMissingLocalQuoteJobs → 실행
function requeueMissingLocalQuoteJobs() {
  let queued = 0;
  const currentYear = new Date().getFullYear();

  for (const dept of Object.keys(CONFIG.DEPARTMENT_ROOTS)) {
    const rootFolderId = CONFIG.DEPARTMENT_ROOTS[dept];
    const years = getAvailableLedgerYears_(rootFolderId, currentYear);

    for (const year of years) {
      const system = getExistingYearSystem_(year, rootFolderId);
      if (!system) continue;

      const sheet = system.ledgerSheet;
      const lastRow = sheet.getLastRow();
      if (lastRow < 2) continue;

      const snapshot = getQuoteSnapshotSheet_(system.ledger, false);
      if (!snapshot || snapshot.getLastRow() < 2) continue;
      const snapshotRows = snapshot.getRange(2, 1, snapshot.getLastRow() - 1, 5).getValues();

      for (let rowIndex = 2; rowIndex <= lastRow; rowIndex++) {
        const quoteNumber = String(sheet.getRange(rowIndex, LEDGER_COL.QUOTE_NUMBER).getDisplayValue() || '').trim();
        const fileLink = String(sheet.getRange(rowIndex, LEDGER_COL.FILE_LINK).getDisplayValue() || '').trim();
        if (!quoteNumber || fileLink) continue;

        const baseQuote = baseQuoteNumber_(quoteNumber);
        const matchingSnapshots = snapshotRows.filter(function(row) {
          return String(row[1] || '').trim() === baseQuote;
        });
        if (matchingSnapshots.length === 0) {
          Logger.log(`[재처리 건너뜀] ${dept}/${year}/${baseQuote}: 원본 스냅샷 없음`);
          continue;
        }

        for (const snapshotRow of matchingSnapshots) {
          let stored;
          try {
            stored = JSON.parse(String(snapshotRow[4] || '{}'));
          } catch (err) {
            Logger.log(`[재처리 실패] ${baseQuote}: 스냅샷 JSON 해석 실패`);
            continue;
          }
          const details = stored.details || {};
          const outputNumber = String(snapshotRow[0] || details.quoteNumber || '').trim();
          if (!outputNumber) continue;

          details.authorDepartment = dept;
          details.year = year;
          details.quoteNumber = outputNumber;
          if (baseQuote !== outputNumber && !details.baseQuoteNumber) details.baseQuoteNumber = baseQuote;

          const pending = getAgentSubfolder_('pending');
          const pendingName = `${sanitizeDriveName(outputNumber)}.json`;
          const existing = pending.getFilesByName(pendingName);
          if (existing.hasNext()) continue;

          enqueueLocalSave_(details, stored.items || [], Number(stored.totalPrice || 0), false, '');
          queued++;
          Logger.log(`[재처리 큐 등록] ${dept}/${year}/${outputNumber}`);
        }
      }
    }
  }

  Logger.log(`대장 누락 파일 재처리 큐 등록 완료: ${queued}건`);
}
